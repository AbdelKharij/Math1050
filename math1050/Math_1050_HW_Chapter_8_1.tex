\begin{problem}
\textbf{1.} Name the greatest common factor of 4, 12, and 16.

$\answer{4}$
\end{problem}

\begin{problem}
\textbf{2.} Name the greatest common factor of $x^3$, $x^5$, and $x^6$.

$\answer{x^3}$
\end{problem}

\begin{problem}
\textbf{75.} \emph{Factor.}\quad $x(a + b) + 2(a + b)$

$\answer{(a + b)(x + 2)}$
\end{problem}

\begin{problem}
\textbf{79.} \emph{Factor.}\quad $a(y - 4) - b(y - 4)$

$\answer{(y - 4)(a - b)}$
\end{problem}

\begin{problem}
\textbf{83.} \emph{Rewrite the expression so that the terms have a common binomial factor.}\quad $b(a - 7) + 3(7 - a)$

$\answer{b(a - 7) - 3(a - 7)}$
\end{problem}

\begin{problem}

$\answer{(x - 2)(a - 5)}$

\begin{problem}

\begin{problem}

$\answer{(3x - 4)(w + 1)}$
\end{problem}

\begin{problem}

\end{problem}

\begin{problem}

\begin{problem}

\begin{problem}
