\begin{problem}
\textbf{93.} Which of the following weather conditions feels \emph{colder}?
  \begin{enumerate}[label=\textbf{\alph*.},leftmargin=2em,topsep=2pt,itemsep=1pt]
    \item a temperature of $5^\circ$F with a $20$ mph wind or a temperature of
          $10^\circ$F with a $15$ mph wind
    \item a temperature of $-25^\circ$F with a $10$ mph wind or a temperature of
          $-15^\circ$F with a $20$ mph wind
  \end{enumerate}

\begin{multipleChoice}
