  \textbf{c.}~$x^2 = 8x$

a.

\begin{multipleChoice} 
   \choice[correct]{Yes} 
   \choice{No}
\end{multipleChoice}

b.

\begin{multipleChoice} 
   \choice{Yes} 
   \choice[correct]{No}
\end{multipleChoice}

c.

\begin{multipleChoice} 
   \choice[correct]{Yes} 
   \choice{No}
\end{multipleChoice}
\end{problem}

\begin{problem}
\textbf{2.} Write the equation in standard form.
  \\[2pt]
  \hspace*{1.5em}
  \textbf{a.}~$x^2 + 4 = 4x$ \qquad \textbf{b.}~$x + x^2 = 6$

a. $\answer{x^2-4x+4=0}$.

b. $\answer{x^2+x-6=0}$.
\end{problem}

\begin{problem}
\textbf{73.} \textbf{Geometry} The length of the base of a triangle is three times the height.
  The area of the triangle is 54 ft$^2$. Find the base and height of the triangle.

The base is $\answer{18}$ feet and the height is $\answer{6}$ feet.
\end{problem}

\begin{problem}
\textbf{75.} \textbf{Geometry} The length of a rectangle is 2 ft more than twice the width. The
  area is 144 ft$^2$. Find the length and width of the rectangle.

The length is $\answer{18}$ feet and the width is $\answer{8}$ feet.
\end{problem}

\begin{problem}
\textbf{77.} \textbf{Geometry} The length of each side of a square is extended 4 m. The area of
  the resulting square is 64 m$^2$. Find the length of a side of the original square.

$\answer{4}$ meters.
\end{problem}

\begin{problem}
\textbf{79.} \textbf{Geometry} The radius of a circle is increased by 3 in., which increases the
  area by 100 in$^2$. Find the radius of the original circle. Round to the nearest
  hundredth.

$\answer{3.81}$ inches.
\end{problem}

\begin{problem}
\textbf{81.} \textbf{Geometry} The page of a book measures 6 in. by 9 in. A uniform border
  around the page leaves 28 in$^2$ for type. What are the dimensions of the type area?

The smaller dimension is $\answer{4}$ inches and the larger dimension is $\answer{7}$ inches.
\end{problem}
