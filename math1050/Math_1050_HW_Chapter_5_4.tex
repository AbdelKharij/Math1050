\begin{problem}
\textbf{1.} The graph of the equation $y = 5x + 7$ has slope \blank{} and $y$-intercept
  $(0, \underline{\hspace{2.2em}})$.

Fill in the blank: $\answer{5}$ and $\answer{7}$
\end{problem}
