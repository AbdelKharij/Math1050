\begin{problem}
\textbf{11.} A computer software retailer uses a markup rate of 40\%. Find the selling price
  of a computer game that costs the retailer \$40.

The selling price is $\answer{56}$ dollars.
\end{problem}

\begin{problem}
\textbf{13.} An electronics store uses a markup rate of 58\%. Find the selling price of a
  camcorder that costs the store owner \$358.

The selling price is \$ $ \answer{565.64}$.
\end{problem}

\begin{problem}
\textbf{15.} A set of golf clubs costing \$360 is sold for \$630. Find the markup rate on the
  set of golf clubs.

The markup rate is $\answer{75}$ percent.
\end{problem}

\begin{problem}
\textbf{17.} A grocer purchases a bottle of fruit juice for \$1.96. The selling price of the
  fruit juice is \$2.45. Find the markup rate.

The markup rate is $\answer{25}$ percent.
\end{problem}

\begin{problem}
\textbf{19.} The selling price of a compact disc player is \$168. The markup rate used by the
  seller is 40\%. Find the cost of the compact disc player.

The cost of the compact disc player is $\answer{120}$ dollars.
\end{problem}

\begin{problem}
\textbf{21.} A markup rate of 40\% was used on a basketball with a selling price of \$82.60.
  Find the cost of the basketball.

The cost of the basketball is $\answer{59}$ dollars.
\end{problem}

\begin{problem}
\textbf{31.} A tennis racket that regularly sells for \$195 is on sale for 25\% off the
  regular price. Find the sale price.

The sale price is \$ $\answer{146.25}$.
\end{problem}

\begin{problem}
\textbf{33.} A supplier of electrical equipment offers a 5\% discount for a purchase that is
  paid for within 30 days. A transformer regularly sells for \$230. Find the discount
  price of a transformer that is paid for 10 days after the purchase.

The discount price is \$ $\answer{218.50}$.
\end{problem}

\begin{problem}
\textbf{35.} A car stereo system that regularly sells for \$425 is on sale for \$318.75. Find
  the discount rate.

The discount rate is $\answer{25}$ percent.
\end{problem}

\begin{problem}
\textbf{37.} A digital camera with a regular price of \$325 is on sale for \$201.50. Find the
  markdown rate.

The markdown rate is $\answer{38}$ percent.
\end{problem}

\begin{problem}
\textbf{39.} The sale price of a free-weight home gym is \$568, which is 20\% off the regular
  price. Find the regular price.

The regular price is $\answer{710}$ dollars.
\end{problem}

\begin{problem}
\textbf{41.} A mechanic's tool set is on sale for \$180 after a markdown of 40\% off the
  regular price. Find the regular price.

The regular price is $\answer{300}$ dollars.
\end{problem}

\begin{problem}
\textbf{45.} A pair of shoes that now sells for \$63 has been marked up 40\%. Find the markup
  on the pair of shoes.

The markup is $\answer{18}$ dollars.
\end{problem}

\begin{problem}
\textbf{47.} A refrigerator selling for \$1087.50 has a markup of \$217.50. Find the markup
  rate.

The markup rate is $\answer{25}$ percent.
\end{problem}

\begin{problem}
\textbf{49.} The manager of a camera store uses a markup rate of 30\%. Find the cost of a
  camera selling for \$299.

The cost of the camera is $\answer{230}$ dollars.
\end{problem}

\begin{problem}
\textbf{51.} A customer buys four tires, three at the regular price and one for 20\% off the
  regular price. The four tires cost \$608. What was the regular price of a tire?

The regular price is $\answer{160}$ dollars.
\end{problem}
