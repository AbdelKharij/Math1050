\begin{problem}
\textbf{15.} In an isosceles triangle, the third side is 50\% of the length of one of the
  equal sides. Find the length of each side when the perimeter is 125 ft.

The longer sides are $\answer{50}$ ft, and the shorter side is $\answer{25}$ ft .
\end{problem}

\begin{problem}
\textbf{17.} The perimeter of a rectangle is 42 m. The length of the rectangle is 3 m less
  than twice the width. Find the length and width of the rectangle.

The length is $\answer{13}$ m . The width is $\answer{8}$ m .
\end{problem}

\begin{problem}
\textbf{19.} The perimeter of a rectangle is 120 ft. The length of the rectangle is twice the
  width. Find the length and width of the rectangle.

The length is $\answer{40}$ ft . The width is $\answer{20}$ ft .
\end{problem}

\begin{problem}
\textbf{21.} The perimeter of a triangle is 110 cm. One side is twice the second side. The
  third side is 30 cm more than the second side. Find the length of each side.

The shortest side is $\answer{20}$ cm, the longest side is $\answer{50}$ cm and the last side is $\answer{40}$ cm.
\end{problem}

\begin{problem}
\textbf{23.} The width of the rectangular foundation of a building is 30\% of the length. The
  perimeter of the foundation is 338 ft. Find the length and width of the foundation.

The length is $\answer{130}$ ft . The width is $\answer{39}$ ft .
\end{problem}

\begin{problem}
\textbf{25.} A rectangular vegetable garden has a perimeter of 64 ft. The length of the garden
  is 20 ft. What is the width of the garden?

The width is $\answer{12}$ ft.
\end{problem}

\begin{problem}
\textbf{27.} The perimeter of a square picture frame is 48 in. Find the length of each side of
  the frame.

Each side measures $\answer{12}$ in .
\end{problem}

  \figbox{%
  \begin{tikzpicture}[scale=0.95]

    \draw[bothray] (0,-1.8) -- (0,2.1);
    \draw[bothray] (-2.3,1.905) -- (2.9,0.085);   % through (0,1.1), slope -0.35

    \coordinate (T) at (0,1.1);
    \coordinate (B) at (0,-0.6);
    \coordinate (R) at (1.889,0.439);
    % angle a: at T, between the upward vertical and the line going down-right
    \draw[arc] ($(T)+(340.71:0.48)$) arc (340.71:450:0.48);
    \node[lbl] at ($(T)+(35.4:0.72)$) {$a$};
    % angle b: at B, between the upward vertical and the line going up-right
    \draw[arc] ($(B)+(28.81:0.48)$) arc (28.81:90:0.48);
    \node[lbl] at ($(B)+(59.4:0.72)$) {$b$};
