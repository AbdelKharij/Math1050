\begin{problem}
\textbf{9.} Traveling with the wind, a plane flies $m$ miles in $h$ hours. Traveling against
  the wind, the plane flies $n$ miles in $h$ hours. Is $m$ less than, equal to, or
  greater than $n$?

\begin{multipleChoice} 
   \choice{Less than} 
   \choice{Equal To} 
   \choice[correct]{Greater than} 
\end{multipleChoice}
\end{problem}

\begin{problem}
\textbf{11.} A whale swimming against an ocean current traveled 60 mi in 2 h. Swimming in the
  opposite direction, with the current, the whale was able to travel the same distance in
  1.5 h. Find the speed of the whale in calm water and the rate of the ocean current.

The speed of the whale in calm water is $\answer{35}$ mph. The rate of the current is $\answer{5}$ mph.
\end{problem}

\begin{problem}
\textbf{13.} A rowing team rowing with the current traveled 40 km in 2 h. Rowing against the
  current, the team could travel only 16 km in 2 h. Find the team's rowing rate in calm
  water and the rate of the current.

The team's rowing rate in calm water was $\answer{14}$ km/h. The rate of the current was $\answer{6}$ km/h.
\end{problem}

\begin{problem}
\textbf{15.} A motorboat traveling with the current went 35 mi in 3.5 h. Traveling against the
  current, the boat went 12 mi in 3 h. Find the rate of the boat in calm water and the
  rate of the current.

The rate of the boat in calm water was $\answer{7}$ mph. The rate of the current was $\answer{3}$ mph.
\end{problem}

\begin{problem}
\textbf{17.} A private Learjet 31A transporting passengers was flying with a tailwind and
  traveled 1120 mi in 2 h. Flying against the wind on the return trip, the jet was able to
  travel only 980 mi in 2 h. Find the speed of the jet in calm air and the rate of the
  wind.

The speed of the jet in calm air was $\answer{525}$ mph. The rate of the wind was $\answer{35}$ mph.
\end{problem}

\begin{problem}
\textbf{19.} A seaplane flying with the wind flew from an ocean port to a lake, a distance of
  240 mi, in 2 h. Flying against the wind, the seaplane made the trip from the lake to the
  ocean port in 3 h. Find the rate of the plane in calm air and the rate of the wind.

The rate of the plane in calm air was $\answer{100}$ mph. The rate of the wind was $\answer{20}$ mph.
\end{problem}

\begin{problem}
\textbf{21.} A Boeing Apache Longbow military helicopter traveling directly into a strong
  headwind was able to travel 450 mi in 2.5 h. The return trip, now with a tailwind, took
  1 h 40 min. Find the speed of the helicopter in calm air and the rate of the wind.

The speed of the helicopter in calm air was $\answer{225}$ mph. The rate of the wind was $\answer{45}$ mph.
\end{problem}

  the spice tea, the mixture costs \$15 per pound. Is the cost per pound of the cinnamon
  tea less than, equal to, or greater than the cost per pound of the spice tea?

\begin{multipleChoice}

\begin{multipleChoice}

\begin{problem}
\textbf{29.} \textbf{Business} A baker purchased 12 lb of wheat flour and 15 lb of rye flour for
  a total cost of \$39.87. A second purchase, at the same prices, included 15 lb of wheat
  flour and 10 lb of rye flour. The cost of the second purchase was \$33.30. Find the cost
  per pound of the wheat and rye flours.

The cost of the wheat is \$ $\answer{.96}$ per pound. The cost of the rye is \$ $\answer{1.89}$ per pound.
\end{problem}

\begin{problem}
\textbf{31.} \textbf{Food Mixtures} A pastry chef created a 50-ounce sugar solution that was
  34\% sugar from a 20\% sugar solution and a 40\% sugar solution. How much of the 20\%
  sugar solution and how much of the 40\% sugar solution were used?

The pastry chef used $\answer{15}$ oz of the 20\% solution and $\answer{35}$ oz of the 40\% solution.
\end{problem}

\begin{problem}
\textbf{39.} \textbf{Investments} An investor has two investments, one earning 5\% annual simple
  interest and the other earning 4.5\% annual simple interest. The two accounts earn \$240
  in interest in one year. If the amounts in each account were reversed, the interest
  earned would be \$235. How much is invested in the 5\% account?

\$ $\answer{3000}$ is invested in the 5\% account.
\end{problem}

\begin{problem}
\textbf{40.} \textbf{Geometry} Two angles are supplementary. The larger angle is $15^\circ$ more
  than twice the measure of the smaller angle. Find the measures of the two angles.
  \emph{(Supplementary angles are two angles whose sum is $180^\circ$.)}

The smaller angle is $\answer{ 55}$ degrees and the larger angle is $\answer{125}$ degrees.
\end{problem}

\begin{problem}
\textbf{41.} \textbf{Geometry} Two angles are complementary. The larger angle is four times the
  measure of the smaller angle. Find the measures of the two angles.
  \emph{(Complementary angles are two angles whose sum is $90^\circ$.)}

The smaller angle is $\answer{ 18}$ degrees and the larger angle is $\answer{72}$ degrees.
\end{problem}

\begin{problem}
\textbf{42.} \textbf{Geometry} The perimeter of a rectangle is 56 cm. The length is 4 cm more
  than three times the width. Find the length of the rectangle.
  \emph{(Use $P = 2L + 2W$.)}

The length is $\answer{22}$ cm.
\end{problem}

\begin{problem}
\textbf{43.} \textbf{Geometry} The perimeter of a rectangle is 68 in. The length is 2 in. less
  than three times the width. Find the length of the rectangle.
  \emph{(Use $P = 2L + 2W$.)}

The length is $\answer{25}$ in.
\end{problem}
