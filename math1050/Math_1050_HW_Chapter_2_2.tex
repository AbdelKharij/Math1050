\begin{problem}
\textbf{37.} Simplify $5a - 8a$ by combining like terms.
  \\[2pt]
  \hspace*{1.5em}$\begin{aligned}
    5a - 8a &= 5a + \underline{\hspace{2.2em}} && \text{Write subtraction as addition of the opposite.}\\
            &= [5 + (-8)]a && \text{Use the \underline{\hspace{2.2em}} Property } ba + ca = (b + c)a.\\
            &= \underline{\hspace{2.2em}}\, a  && \text{Add the coefficients.}
  \end{aligned}$

\textbf{Answer:} $\answer{-8a}$ and $\answer{-3a}$
\end{problem}
