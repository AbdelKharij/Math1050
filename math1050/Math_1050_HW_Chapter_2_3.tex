\begin{problem}

\textbf{Answer:} $\answer{25-x}$

\begin{problem}
\textbf{93.} \textbf{Genetics} The human genome contains 11,000 more genes than the roundworm
  genome. Express the number of genes in the human genome in terms of the number of
  genes in the roundworm genome. \emph{(Source: Celera, USA TODAY research)}

\textbf{Answer:} $\answer{g+11000}$
\end{problem}

\begin{problem}
\textbf{97.} \textbf{Sports} In football, the number of points awarded for a touchdown is
  three times the number of points awarded for a safety. Express the number of points
  awarded for a touchdown in terms of the number of points awarded for a safety.

\textbf{Answer:} $\answer{3s}$
\end{problem}

\begin{problem}
\textbf{101.} \textbf{Geometry} In a triangle, the measure of the smallest angle is 10 degrees
  less than one-half the measure of the largest angle. Express the measure of the
  smallest angle in terms of the measure of the largest angle.

\textbf{Answer:} $\answer{\frac{1}{2}L-10}$
\end{problem}

\begin{problem}
\textbf{105.} \textbf{Coins} A coin bank contains thirty-five coins in nickels and dimes. Use
  the same variable to express the number of nickels and the number of dimes in the
  coin bank.

\textbf{Answer:} $\answer{n}$ and $\answer{35-n}$
\end{problem}
