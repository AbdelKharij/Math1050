When entering fractions, make sure the denominator is inside parentheses (from order of operations).

\section*{Section 7.5}

\noindent
\textbf{9.} \emph{Divide.}\quad $\dfrac{10a - 25}{5}$

$\answer{2a - 5}$

\noindent
\textbf{11.} \emph{Divide.}\quad $\dfrac{3a^2 + 2a}{a}$

$\answer{3a + 2}$

\noindent
\textbf{13.} \emph{Divide.}\quad $\dfrac{3x^2 - 6x}{3x}$

$\answer{x - 2}$

\noindent
\textbf{15.} \emph{Divide.}\quad $\dfrac{5x^2 - 10x}{-5x}$

$\answer{-x + 2}$

\noindent
\textbf{17.} \emph{Divide.}\quad $\dfrac{x^3 + 3x^2 - 5x}{x}$

$\answer{x^2 + 3x - 5}$

\noindent
\textbf{19.} \emph{Divide.}\quad $\dfrac{x^6 - 3x^4 - x^2}{x^2}$

$\answer{x^4 - 3x^2 - 1}$

\noindent
\textbf{21.} \emph{Divide.}\quad $\dfrac{5x^2y^2 + 10xy}{5xy}$

$\answer{xy + 2}$

\noindent
\textbf{23.} \emph{Divide.}\quad $\dfrac{9y^6 - 15y^3}{-3y^3}$

$\answer{-3y^3 + 5}$

\noindent
\textbf{25.} \emph{Divide.}\quad $\dfrac{3x^2 - 2x + 1}{x}$

$\answer{3x - 2 + \frac{1}{x}}$

\noindent
\textbf{27.} \emph{Divide.}\quad $\dfrac{-3x^2 + 7x - 6}{x}$

$\answer{-3x + 7 - \frac{6}{x}}$

\noindent
\textbf{29.} \emph{Divide.}\quad $\dfrac{16a^2b - 20ab + 24ab^2}{4ab}$

$\answer{4a - 5 + 6b}$

\noindent
\textbf{31.} \emph{Divide.}\quad $\dfrac{9x^2y + 6xy - 3x}{xy}$

$\answer{9x + 6 - \frac{3}{y}}$

\noindent
\textbf{35.} \emph{Divide.}\quad $(x^2 - x - 6) \div (x - 3)$

$\answer{x + 2}$

\noindent
\textbf{37.} \emph{Divide.}\quad $(2x^2 + 5x + 2) \div (x + 2)$

$\answer{2x + 1}$

\noindent
\textbf{39.} \emph{Divide.}\quad $(x^2 + 1) \div (x - 1)$

$\answer{x + 1 + \frac{2}{x-1}}$

\noindent
\textbf{41.} \emph{Divide.}\quad $(6x^2 - 7x) \div (3x - 2)$

$\answer{2x - 1 - \frac{2}{3x-2}}$

\noindent
\textbf{43.} \emph{Divide.}\quad $(b^2 - 8b - 9) \div (b - 3)$

$\answer{b - 5 - \frac{24}{b-3}}$

\noindent
\textbf{45.} \emph{Divide.}\quad $(3x^2 + 5x - 4) \div (x - 4)$

$\answer{3x + 17 + \frac{64}{x-4}}$

\noindent
\textbf{47.} \emph{Divide.}\quad $(10 + 21y + 10y^2) \div (2y + 3)$

$\answer{5y + 3 + \frac{1}{2y+3}}$

\noindent
\textbf{49.} \emph{Divide.}\quad $(5 - 23x + 12x^2) \div (4x - 1)$

$\answer{3x - 5}$

\noindent
\textbf{51.} \emph{Divide.}\quad $(3x^2 + x^3 + 8 + 5x) \div (x + 1)$

$\answer{x^2 + 2x + 3 + \frac{5}{x+1}}$

\noindent
\textbf{53.} \emph{Divide.}\quad $(x^4 - x^2 - 6) \div (x^2 + 2)$

$\answer{x^2 - 3}$
