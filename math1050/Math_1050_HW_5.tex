\section*{A rectangle with semicircular ends}

\figbox{\begin{tikzpicture}[x=0.038cm,y=0.038cm]
  \draw[edge] (0,40) -- (70,40);
  \draw[edge] (0,0)  -- (70,0);
  \draw[edge] (0,20) ++(90:20) arc (90:270:20);
  \draw[edge] (70,20) ++(-90:20) arc (-90:90:20);
  \draw[dim] (0,-13) -- (70,-13);
  \node[lbl] at (35,-13) {70 m};
  \draw[dim] (102,0) -- (102,40);
  \node[lbl] at (102,20) {40 m};
  \draw[dimln] (0,0) -- (0,-16);
  \draw[dimln] (70,0) -- (70,-16);
  \draw[dimln] (70,0) -- (105,0);
  \draw[dimln] (70,40) -- (105,40);
\end{tikzpicture}}

\begin{problem}
   Find the perimeter (in meters) of the figure above. Use 3.14 for \(\pi\).

\[
\text{ The perimeter is } \answer{265.6} \text{ meters }.
\]

\begin{hint}
\textbf{Solution.} The two straight sides contribute $70+70=140$ m. The two semicircular
ends together make one full circle of diameter 40 m, contributing
$\pi d = 3.14(40) = 125.6$ m. The perimeter is $140 + 125.6 = 265.6$ m.
\end{hint}
\end{problem}

\begin{problem}
   Find the area (in square meters) of the figure above. Use 3.14 for \(\pi\).

   \[
   \text{ The area is } \answer{4056} \text{ square meters}.
   \]

\begin{hint}
\textbf{Solution.} The rectangle contributes $70(40) = 2800$ m$^2$. The two semicircular
ends together make one full circle of radius 20 m, contributing
$\pi r^2 = 3.14(20)^2 = 1256$ m$^2$. The area is $2800 + 1256 = 4056$ m$^2$.
\end{hint}
\end{problem}

\section*{A square border}

The shaded region is the border between the two squares.

\figbox{\begin{tikzpicture}[x=0.032cm,y=0.032cm]
  \fill[fillblue] (0,0) rectangle (85,85);
  \fill[white] (5,5) rectangle (80,80);
  \draw[edge] (0,0) rectangle (85,85);
  \draw[edge] (5,5) rectangle (80,80);
  \draw[dim] (0,-15) -- (85,-15);
  \node[lbl] at (42.5,-15) {85 m};
  \draw[dim] (100,5) -- (100,80);
  \node[lbl] at (100,42.5) {75 m};
  \draw[dimln] (0,0) -- (0,-18);
  \draw[dimln] (85,0) -- (85,-18);
  \draw[dimln] (80,5)  -- (103,5);
  \draw[dimln] (80,80) -- (103,80);
\end{tikzpicture}}

\begin{problem}
Find the outside perimeter (in meters) of the shaded region of the figure above.

\[
\text{ The perimeter is } \answer{340} \text{ meters }.
\]

\begin{hint}
\textbf{Solution.} The outside boundary is the larger square, whose side is 85 m, so the
outside perimeter is $4(85) = 340$ m.
\end{hint}
\end{problem}

\begin{problem}
   Find the area (in square meters) of the shaded region of the figure above.

   \[
   \text{ The area is } \answer{1600} \text{ square meters}.
   \]

\begin{hint}
\textbf{Solution.} Subtract the smaller square from the larger one:
$85^2 - 75^2 = 7225 - 5625 = 1600$ m$^2$.
\end{hint}
\end{problem}
