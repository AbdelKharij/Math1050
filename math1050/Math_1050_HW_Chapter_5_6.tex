\begin{problem}
\textbf{9.} \emph{Graph the solution set.}\quad $y > 2x + 3$              \figbox{\gridsm}

\begin{hint}
\textbf{Solution.} Solved for $y$ the equation is $y=2x + 3$, so the slope is $2$ and the $y$-intercept is $(0,3)$. Two more lattice points on the line are $(-1,1)$ and $(1,5)$. Draw that boundary line dashed (points on it are excluded), then test $(0,0)$: it does not satisfy the inequality, so shade the half-plane on the other side from $(0,0)$.
\end{hint}
\end{problem}
