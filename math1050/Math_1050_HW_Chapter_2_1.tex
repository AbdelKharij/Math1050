\maketitle

\begin{problem}
\textbf{9.} \emph{Name the terms of the variable expression. Then underline the constant term.}\quad $-3n^2 - 4n + 7$
\end{problem}

\begin{problem}
\textbf{13.} \emph{Name the variable terms of the expression. Then underline the variable part of each term.}\quad $5 - 8n - 3n^2$
\end{problem}

\begin{problem}
\textbf{17.} What is the meaning of the phrase ``evaluate a variable expression''?
\end{problem}

\begin{problem}
\textbf{21.} \emph{Evaluate the variable expression when $a = 2$, $b = 3$, and $c = -4$.}\quad $-a^2$

\textbf{Answer:} $\answer{-4}$
\end{problem}

\begin{problem}
\textbf{25.} \emph{Evaluate the variable expression when $a = 2$, $b = 3$, and $c = -4$.}\quad $b^2 - 3$

\textbf{Answer:} $\answer{6}$
\end{problem}

\begin{problem}
\textbf{29.} \emph{Evaluate the variable expression when $a = 2$, $b = 3$, and $c = -4$.}\quad $bc \div (2a)$

\textbf{Answer:} $\answer{-3}$
\end{problem}

\begin{problem}
\textbf{33.} \emph{Evaluate the variable expression when $a = 2$, $b = 3$, and $c = -4$.}\quad $(a + b)^2$

\textbf{Answer:} $\answer{25}$
\end{problem}

\begin{problem}
\textbf{37.} \emph{Evaluate the variable expression when $a = 2$, $b = 3$, and $c = -4$.}\quad $b^2 - \dfrac{ac}{8}$

\textbf{Answer:} $\answer{10}$
\end{problem}

\begin{problem}
\textbf{41.} \emph{Evaluate the variable expression when $a = -2$, $b = 4$, $c = -1$, and $d = 3$.}\quad $\dfrac{d - b}{c}$

\textbf{Answer:} $\answer{1}$
\end{problem}

\begin{problem}
\textbf{45.} \emph{Evaluate the variable expression when $a = -2$, $b = 4$, $c = -1$, and $d = 3$.}\quad $\dfrac{2c - d}{-ad}$

\textbf{Answer:} $\answer{-\frac{5}{6}}$
\end{problem}

\begin{problem}
\textbf{49.} \emph{Evaluate the variable expression when $a = -2$, $b = 4$, $c = -1$, and $d = 3$.}\quad $(b - c)^2 \div 5$

\textbf{Answer:} $\answer{5}$
\end{problem}

\begin{problem}
\textbf{80.} \emph{Evaluate the variable expression when $a = -2$ and $b = -3$.}\quad $|2a + 3b|$

\textbf{Answer:} $\answer{13}$
\end{problem}

\begin{problem}
\textbf{81.} \emph{Evaluate the variable expression when $a = -2$ and $b = -3$.}\quad $\lvert -4ab \rvert$

\textbf{Answer:} $\answer{24}$
\end{problem}
