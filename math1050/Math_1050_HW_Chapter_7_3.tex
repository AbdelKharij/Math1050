\begin{problem}
\textbf{133.} \textbf{Geometry} The width of a rectangle is $(x - 6)$ m. The length is
  $(2x + 3)$ m. Find the area of the rectangle in terms of the variable $x$.
  \figbox{%
  \begin{tikzpicture}[scale=0.85]
    \fill[figlavender] (0,0) rectangle (2.6,1.15);
    \draw[edge] (0,0) rectangle (2.6,1.15);
    \node[lbl,above] at (1.3,1.15) {$2x + 3$};
    \node[lbl,right] at (2.6,0.575) {$x - 6$};
  \end{tikzpicture}}

The area is $\answer{2x^2 - 9x - 18}$ square meters.
\end{problem}
