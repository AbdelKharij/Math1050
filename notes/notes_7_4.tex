\section*{Rules for exponents}

\textbf{A.} $x^{m}\cdot x^{n}=x^{m+n}$. For example $2^{3}\cdot 2^{4}=2^{7}$, because
$2\cdot 2\cdot 2\cdot 2\cdot 2\cdot 2\cdot 2$ is seven factors of 2.

\textbf{B.} $\left(x^{m}\right)^{n}=x^{mn}$. For example
$\left(2^{3}\right)^{4}=2^{12}$, since $2^{3}$ is multiplied by itself four times.

\textbf{C.} $\left(x^{m}y^{n}\right)^{p}=x^{mp}y^{np}$. For example
$\left(2^{3}\cdot 3^{4}\right)^{5}=2^{15}\cdot 3^{20}$.

\textbf{D.} $\dfrac{x^{m}}{x^{n}}=x^{m-n}$ for $x \neq 0$. For example
$\dfrac{x^{4}}{x^{2}}=x^{2}$.

\textbf{E.} $x^{-n}=\dfrac{1}{x^{n}}$ for $x \neq 0$. For example
$\dfrac{x^{3}}{x^{5}}=x^{-2}=\dfrac{1}{x^{2}}$.

\textbf{F.} $x^{0}=1$ for $x \neq 0$, since $\dfrac{x^{5}}{x^{5}}=x^{0}$ and any nonzero
number divided by itself is 1.

\section*{Worked examples}

Simplify, writing answers without negative exponents.

\begin{problem}

\textbf{1.} $\left(\dfrac{a^{7}}{b^{11}}\right)^{-3}$

\begin{hint}
\textbf{Answer.} $\dfrac{b^{33}}{a^{21}}$.
\end{hint}
\end{problem}

\begin{problem}

\textbf{2.} $\dfrac{-a^{-10}}{b^{12}}$

\begin{hint}
\textbf{Answer.} $-\dfrac{1}{a^{10}b^{12}}$.
\end{hint}
\end{problem}

\begin{problem}

\textbf{3.} $\dfrac{q}{q^{-8}}$

\begin{hint}
\textbf{Answer.} $q^{1-(-8)}=q^{9}$.
\end{hint}
\end{problem}

\begin{problem}

\textbf{4.} $4^{-2}-4^{-3}$

\begin{hint}
\textbf{Answer.} $\dfrac{1}{16}-\dfrac{1}{64}=\dfrac{4}{64}-\dfrac{1}{64}=\dfrac{3}{64}$.
\end{hint}
\end{problem}

\begin{problem}

\textbf{5.} $\dfrac{-1}{x^{-13}}$

\begin{hint}
\textbf{Answer.} $-x^{13}$.
\end{hint}
\end{problem}

\begin{problem}

\textbf{6.} $\dfrac{8x^{10}}{-4x^{11}}$

\begin{hint}
\textbf{Answer.} $\dfrac{8}{-4}\cdot\dfrac{x^{10}}{x^{11}} = -\dfrac{2}{x}$.
\end{hint}
\end{problem}

\begin{problem}

\textbf{7.} Write the number $4.868 \times 10^{-7}$ in standard notation.

\begin{hint}
\textbf{Answer.} $0.0000004868$.
\end{hint}
\end{problem}

\begin{problem}

\textbf{8.} Multiply or divide.

(a) $\left(3.0 \times 10^{5}\right)\left(1.1 \times 10^{-8}\right)$

(b) $\dfrac{7.2 \times 10^{13}}{2.4 \times 10^{-3}}$

\begin{hint}
\textbf{Answer.} (a) $(3.0)(1.1)\times 10^{5+(-8)} = 3.3 \times 10^{-3}$.
(b) $\dfrac{7.2}{2.4}\times 10^{13-(-3)} = 3.0 \times 10^{16}$.
\end{hint}
\end{problem}

\begin{problem}

\textbf{10.} $\dfrac{5r^{3}t^{7}}{6r^{5}t^{7}}$ \qquad

\begin{hint}
\textbf{Answer.}
$\dfrac{5}{6}r^{3-5}t^{7-7}=\dfrac{5}{6r^{2}}$.
\end{hint}
\end{problem}

\textbf{11.} $\left(-2xy^{-2}\right)^{2}$ \qquad

\begin{hint}
\textbf{Answer.}
$(-2)^{2}x^{2}y^{-4}=\dfrac{4x^{2}}{y^{4}}$.
\end{hint}
\end{problem}
