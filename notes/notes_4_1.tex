\begin{abstract}
    Section 4.1 --- Translating Sentences into Equations.
    Worked examples, then practice problems with answers.
\end{abstract}

\maketitle

Translate the sentence into an equation, then solve it. Words such as \emph{is} and
\emph{was} become the equals sign; \emph{sum}, \emph{difference}, \emph{product} and
\emph{quotient} become $+$, $-$, $\cdot$ and $\div$.

For consecutive integers use $n$, $n+1$, $n+2$; for consecutive even or odd integers use
$n$, $n+2$, $n+4$.

\section*{Translate and solve}

\begin{problem}
\textbf{1.} The sum of five and a number is three. Find the number.
\qquad \textbf{Answer:} $\answer{-2}$

\begin{hint}
\textbf{Solution.} $5+x=3$, so $x=-2$.
\end{hint}
\end{problem}

\begin{problem}
\textbf{2.} The quotient of a number and four is two. Find the number.
\qquad \textbf{Answer:} $\answer{8}$

\begin{hint}
\textbf{Solution.} $\dfrac{x}{4}=2$, so $x=8$.
\end{hint}
\end{problem}

\begin{problem}
\textbf{3.} The difference between five and twice a number is one. Find the number.
\qquad \textbf{Answer:} $\answer{2}$

\begin{hint}
\textbf{Solution.} $5-2x=1$, so $-2x=-4$ and $x=2$.
\end{hint}
\end{problem}

\begin{problem}
\textbf{4.} Twenty-one is three times the difference between four times a number and five.
Find the number. \qquad \textbf{Answer:} $\answer{3}$

\begin{hint}
\textbf{Solution.} $21=3(4x-5)=12x-15$, so $36=12x$ and $x=3$.
\end{hint}
\end{problem}

\begin{problem}
\textbf{5.} Negative fifteen is three more than twice a number. Find the number.
\qquad \textbf{Answer:} $\answer{-9}$

\begin{hint}
\textbf{Solution.} $-15=2x+3$, so $2x=-18$ and $x=-9$.
\end{hint}
\end{problem}

\begin{problem}
\textbf{6.} The difference between three times a number and four is five times the number.
Find the number. \qquad \textbf{Answer:} $\answer{-2}$

\begin{hint}
\textbf{Solution.} $3x-4=5x$, so $-4=2x$ and $x=-2$.
\end{hint}
\end{problem}

\begin{problem}
\textbf{7.} The difference between six times a number and four times the number is
negative fourteen. Find the number. \qquad \textbf{Answer:} $\answer{-7}$

\begin{hint}
\textbf{Solution.} $6x-4x=-14$, so $2x=-14$ and $x=-7$.
\end{hint}
\end{problem}

\section*{Consecutive integers}

\begin{problem}
\textbf{8.} The sum of three consecutive integers is 75. Find the integers.

\textbf{Answer:} $\answer{24}$, $\answer{25}$ and $\answer{26}$

\begin{hint}
\textbf{Solution.} $n+(n+1)+(n+2)=3n+3=75$, so $n=24$.
\end{hint}
\end{problem}

\begin{problem}
\textbf{9.} The sum of three consecutive even integers is 48. Find the integers.

\textbf{Answer:} $\answer{14}$, $\answer{16}$ and $\answer{18}$

\begin{hint}
\textbf{Solution.} $n+(n+2)+(n+4)=3n+6=48$, so $n=14$.
\end{hint}
\end{problem}

\begin{problem}
\textbf{10.} The sum of three consecutive odd integers is 81. Find the integers.

\textbf{Answer:} $\answer{25}$, $\answer{27}$ and $\answer{29}$

\begin{hint}
\textbf{Solution.} $3n+6=81$, so $n=25$.
\end{hint}
\end{problem}

\begin{problem}
\textbf{11.} Find three consecutive even integers whose sum is negative twelve.

\textbf{Answer:} $\answer{-6}$, $\answer{-4}$ and $\answer{-2}$

\begin{hint}
\textbf{Solution.} $3n+6=-12$, so $n=-6$.
\end{hint}
\end{problem}

\section*{Applications}

\begin{problem}
\textbf{13.} The length of the Royal Gorge Bridge in Colorado is 320 m. This is
one-fourth the length of the Golden Gate Bridge. Find the length of the Golden Gate
Bridge. \qquad \textbf{Answer:} $\answer{1280}$ metres

\begin{hint}
\textbf{Solution.} $320=\dfrac{1}{4}L$, so $L=1280$ m.
\end{hint}
\end{problem}

\begin{problem}
\textbf{14.} Greek architects considered a rectangle whose length was approximately 1.6
times its width to be the most visually appealing, and called it a Golden Rectangle. Find
the length and width of a Golden Rectangle if the sum of the length and width is 130 feet.

\textbf{Answer:} length $\answer{80}$ ft and width $\answer{50}$ ft

\begin{hint}
\textbf{Solution.} With $w$ the width, $w+1.6w=2.6w=130$, so $w=50$ ft and the length is
$1.6(50)=80$ ft.
\end{hint}
\end{problem}

\begin{problem}
\textbf{15.} A union charges monthly dues of \$4.00 plus \$0.25 for each hour worked
during the month. A member's dues for March were \$46.00. How many hours did the member
work? \qquad \textbf{Answer:} $\answer{168}$ hours

\begin{hint}
\textbf{Solution.} $4+0.25h=46$, so $0.25h=42$ and $h=168$.
\end{hint}
\end{problem}

\section*{Your turn}

\begin{problem}
\textbf{1.} The sum of a number and six is four less than six times the number. Find the
number. \qquad \textbf{Answer:} $\answer{2}$

\begin{hint}
\textbf{Solution.} $x+6=6x-4$, so $10=5x$ and $x=2$.
\end{hint}
\end{problem}

\begin{problem}
\textbf{2.} The sum of three consecutive integers is 54. Find the integers.

\textbf{Answer:} $\answer{17}$, $\answer{18}$ and $\answer{19}$
\end{problem}

\begin{problem}
\textbf{3.} The sum of three consecutive even integers is 84. Find the integers.

\textbf{Answer:} $\answer{26}$, $\answer{28}$ and $\answer{30}$
\end{problem}

\begin{problem}
\textbf{4.} Find two consecutive even integers such that five times the first integer is
equal to four times the second integer.

\textbf{Answer:} $\answer{8}$ and $\answer{10}$

\begin{hint}
\textbf{Solution.} $5n=4(n+2)$, so $n=8$ and the integers are 8 and 10.
\end{hint}
\end{problem}

\begin{problem}
\textbf{5.} One slice of cheese pizza contains 290 calories. A medium-size orange has
one-fifth that number of calories. How many calories are in a medium-size orange?
\qquad \textbf{Answer:} $\answer{58}$

\begin{hint}
\textbf{Solution.} $\dfrac{290}{5}=58$ calories.
\end{hint}
\end{problem}

\begin{problem}
\textbf{6.} Greenland, the largest island in the world, is 21 times larger than Iceland.
The combined area of Greenland and Iceland is $880{,}000$ mi$^{2}$. Find the area of
Greenland. \qquad \textbf{Answer:} $\answer{840000}$ square miles

\begin{hint}
\textbf{Solution.} With $I$ the area of Iceland, $I+21I=22I=880{,}000$, so $I=40{,}000$
and Greenland is $21(40{,}000)=840{,}000$ mi$^{2}$.
\end{hint}
\end{problem}
