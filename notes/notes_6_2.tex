\begin{abstract}
    Section 6.2 --- Solving Systems of Linear Equations by Substitution.
    Worked examples, then practice problems with answers.
\end{abstract}

\maketitle

To use substitution, first write one of the equations in terms of $x$ or in terms of
$y$, that is $x=\cdots$ or $y=\cdots$. Then substitute that expression into the other
equation.

\textbf{Note.} A false result such as $2=5$ means the system has \textbf{no solution}. A
result that is always true, such as $6=6$, means there are \textbf{infinitely many
solutions}.

\section*{Worked examples}

\begin{problem}
\textbf{Example 1.} Solve by substitution.
\[
  \left\{\begin{aligned}
    2x+5y &= 11 \\
    y &= 3x-9
  \end{aligned}\right.
\]

\textbf{Solution.} Substitute $y=3x-9$ into the first equation:
\[
  2x+5(3x-9)=11 \;\Longrightarrow\; 17x-45=11 \;\Longrightarrow\; 17x=56
  \;\Longrightarrow\; x=\tfrac{56}{17}\approx 3.29 .
\]
Then $y=3\left(\tfrac{56}{17}\right)-9=\tfrac{15}{17}\approx 0.88$, so the solution is
$\left(\tfrac{56}{17},\tfrac{15}{17}\right)\approx(3.29,\,0.88)$.
\end{problem}

\begin{problem}
\textbf{Example 2.} Solve by substitution.
\[
  \left\{\begin{aligned}
    y &= -5x+4 \\
    2x-3y &= 5
  \end{aligned}\right.
\]

\textbf{Solution.} Substitute $y=-5x+4$ into the second equation:
\[
  2x-3(-5x+4)=5 \;\Longrightarrow\; 2x+15x-12=5 \;\Longrightarrow\; 17x=17
  \;\Longrightarrow\; x=1 .
\]
Then $y=-5(1)+4=-1$, so the solution is $(1,-1)$.
\end{problem}

\begin{problem}
\textbf{Example 3.} Solve by substitution.
\[
  \left\{\begin{aligned}
    x &= y+1 \\
    x+2y &= 7
  \end{aligned}\right.
\]

\textbf{Solution.} Substitute $x=y+1$:
\[
  (y+1)+2y=7 \;\Longrightarrow\; 3y+1=7 \;\Longrightarrow\; 3y=6 \;\Longrightarrow\; y=2 .
\]
Then $x=2+1=3$, so the solution is $(3,2)$.
\end{problem}

\section*{Your turn}

\begin{problem}
\textbf{1.} Solve by substitution.
\[
  \left\{\begin{aligned}
    2x+3y &= 9 \\
    y &= x+4
  \end{aligned}\right.
\]

\textbf{Answer.} $\left(-\tfrac{3}{5},\ \tfrac{17}{5}\right)$.
\end{problem}

\begin{problem}
\textbf{2.} Solve by substitution.
\[
  \left\{\begin{aligned}
    x-7y &= 4 \\
    -3x+2y &= 6
  \end{aligned}\right.
\]

\textbf{Answer.} $\left(-\tfrac{50}{19},\ -\tfrac{18}{19}\right)$.
\end{problem}

\begin{problem}
\textbf{3.} Solve by substitution.
\[
  \left\{\begin{aligned}
    6x-3y &= 6 \\
    2x-y &= 2
  \end{aligned}\right.
\]

\textbf{Answer.} The second equation is the first divided by 3, so substituting gives
$6=6$: there are \textbf{infinitely many solutions}, every point on $y=2x-2$.
\end{problem}

\begin{problem}
\textbf{4.} Solve by substitution.
\[
  \left\{\begin{aligned}
    y &= 2x+11 \\
    y &= 5x-1
  \end{aligned}\right.
\]

\textbf{Answer.} $(4,\,19)$.
\end{problem}

\begin{problem}
\textbf{5.} Solve by substitution.
\[
  \left\{\begin{aligned}
    4x+3y &= 0 \\
    2x-y &= 0
  \end{aligned}\right.
\]

\textbf{Answer.} $(0,\,0)$.
\end{problem}

\begin{problem}
\textbf{6.} Solve by substitution.
\[
  \left\{\begin{aligned}
    x &= y-2 \\
    3x-2y &= 6
  \end{aligned}\right.
\]

\textbf{Answer.} $(10,\,12)$.
\end{problem}
