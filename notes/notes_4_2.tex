\begin{abstract}
    Section 4.2 --- Geometry: angles formed by intersecting and parallel lines.
    Four figures with answers.
\end{abstract}

\maketitle

When two parallel lines are cut by a transversal, every angle formed is equal either to
the acute angle or to its supplement. Vertical angles are equal, and angles that together
form a straight line add to $180^{\circ}$.

\begin{problem}
\textbf{1.} In Figure 1, lines $f$ and $g$ are parallel. Fill in all the unlabeled angles.
\begin{image}
\includegraphics{1cd8e80f-0c30-4a46-9c14-3f0e580d66f7-1_1089_1828_543_173}
\end{image}

\begin{hint}
\textbf{Answer.} The marked angle is $40^{\circ}$, so every angle in the figure is either
$40^{\circ}$ or $180^{\circ}-40^{\circ}=140^{\circ}$. At each of the two intersections the
angles run $40^{\circ}$, $140^{\circ}$, $40^{\circ}$, $140^{\circ}$ around the point, and
the pattern at $g$ matches the pattern at $f$ because the lines are parallel.
\end{hint}
\end{problem}

\begin{problem}
\textbf{2.} In Figure 2, lines $f$ and $g$ are parallel. Find the measure of angle $x$.
\begin{image}
\includegraphics{1cd8e80f-0c30-4a46-9c14-3f0e580d66f7-1_552_913_2008_640}
\end{image}

\textbf{Answer:} $\answer{160}$ degrees.

\begin{hint}
\textbf{Solution.} The $160^{\circ}$ angle sits above $f$ and left of the transversal.
Its corresponding angle at $g$ is above $g$ and left of the transversal, and $x$ is
vertical to that one, so $x=160^{\circ}$.
\end{hint}
\end{problem}

\begin{problem}
\textbf{3.} In Figure 3, find the value of $x$.
\begin{image}
\includegraphics{1cd8e80f-0c30-4a46-9c14-3f0e580d66f7-2_540_916_361_654}
\end{image}

\textbf{Answer:} $\answer{12}$

\begin{hint}
\textbf{Solution.} $7x$ and $4x+36$ are vertical angles, so they are equal:
\[
  7x=4x+36 \;\Longrightarrow\; 3x=36 \;\Longrightarrow\; x=12 .
\]
\end{hint}
\end{problem}

\begin{problem}
\textbf{4.} In Figure 4, find the value of $x$.
\begin{image}
\includegraphics{1cd8e80f-0c30-4a46-9c14-3f0e580d66f7-2_318_913_1319_651}
\end{image}

\textbf{Answer:} $\answer{18}$

\begin{hint}
\textbf{Solution.} The three angles together form a straight line, so they add to
$180^{\circ}$:
\[
  3x+(x+36)+4x=180 \;\Longrightarrow\; 8x+36=180 \;\Longrightarrow\; x=18 .
\]
\end{hint}
\end{problem}
