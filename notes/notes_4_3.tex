\section*{Definitions}

\begin{itemize}
  \item \textbf{Cost} --- the price a business pays for an item.
  \item \textbf{Selling price} --- the price for which a business sells an item.
  \item \textbf{Markup} --- the difference between the selling price and the cost.
  \item \textbf{Markup rate} --- a percentage of the retailer's cost.
  \item \textbf{Discount} --- the amount a retailer reduces the selling price of a product.
  \item \textbf{Discount rate} (or markdown rate) --- a percentage of the regular price.
\end{itemize}

\textbf{Markup:} Selling price $=$ Cost $+$ Markup $=$ Cost $+$ rate $\cdot$ Cost, that is
$S=C+rC$.

\textbf{Discount:} Sale price $=$ Regular price $-$ Discount $=$ Regular price $-$ rate
$\cdot$ Regular price, that is $S=R-rR$.

\section*{Worked examples}

\begin{problem}
\textbf{1.} A computer software retailer uses a markup rate of $40\%$. Find the selling
price of a computer game that costs the retailer \$40.
\qquad \textbf{Answer:} $\answer{56}$

\begin{hint}
\textbf{Solution.} $S=C+rC=40+0.4(40)=40+16=\$56$.
\end{hint}
\end{problem}

\begin{problem}
\textbf{2.} A car dealer advertises a $5\%$ markup over cost. Find the selling price of a
car that costs the dealer \$26,000. \qquad \textbf{Answer:} $\answer{27300}$

\begin{hint}
\textbf{Solution.} $S=26{,}000+0.05(26{,}000)=26{,}000+1300=\$27{,}300$.
\end{hint}
\end{problem}

\begin{problem}
\textbf{3.} The cost to a landscape architect for a 25-gallon tree is \$65. Find the
selling price if the markup rate is $30\%$. \qquad \textbf{Answer:} $\answer{84.50}$

\begin{hint}
\textbf{Solution.} $S=65+0.3(65)=65+19.50=\$84.50$.
\end{hint}
\end{problem}

\begin{problem}
\textbf{4.} A jeweler purchases a diamond ring for \$7198.50. The selling price is
\$14,397. Find the markup rate. \qquad \textbf{Answer:} $\answer{100}$ percent

\begin{hint}
\textbf{Solution.} $14{,}397=7198.50+r(7198.50)$, so $r(7198.50)=7198.50$ and $r=1=100\%$.
\end{hint}
\end{problem}

\begin{problem}
\textbf{5.} A sofa costing \$520 is sold for \$779. Find the markup rate, rounded to the
nearest tenth of a percent. \qquad \textbf{Answer:} $\answer{49.8}$ percent

\begin{hint}
\textbf{Solution.} $779=520+r(520)$, so $r=\dfrac{259}{520}=0.498=49.8\%$.
\end{hint}
\end{problem}

\begin{problem}
\textbf{6.} A manufacturer uses a markup rate of $45\%$. A treadmill has a selling price
of \$870. Find the cost. \qquad \textbf{Answer:} $\answer{600}$

\begin{hint}
\textbf{Solution.} $870=C+0.45C=1.45C$, so $C=\dfrac{870}{1.45}=\$600$.
\end{hint}
\end{problem}

\begin{problem}
\textbf{7.} A game that costs \$13 is marked up $24\%$. What is the selling price?
\qquad \textbf{Answer:} $\answer{16.12}$

\begin{hint}
\textbf{Solution.} $S=13+0.24(13)=13+3.12=\$16.12$.
\end{hint}
\end{problem}

\begin{problem}
\textbf{8.} A fax machine that regularly sells for \$219.99 is on sale for
$33\frac{1}{3}\%$ off the regular price. Find the sale price.
\qquad \textbf{Answer:} $\answer{146.66}$

\begin{hint}
\textbf{Solution.} A third off leaves two thirds:
$219.99\left(\dfrac{2}{3}\right)=\$146.66$.
\end{hint}
\end{problem}

\begin{problem}
\textbf{9.} A clothing wholesaler offers a discount of $10\%$ per shirt when 10 to 20
shirts are purchased, and $15\%$ per shirt when 21 to 50 shirts are purchased. A shirt
regularly sells for \$27. Find the sale price per shirt when 35 shirts are purchased.
\qquad \textbf{Answer:} $\answer{22.95}$

\begin{hint}
\textbf{Solution.} 35 shirts falls in the 21--50 band, so the discount is $15\%$:
$27-0.15(27)=27(0.85)=\$22.95$.
\end{hint}
\end{problem}

\begin{problem}
\textbf{10.} A pair of roller blades that regularly sells for \$99.99 is on sale for
\$79.99. Find the discount rate, rounded to the nearest percent.
\qquad \textbf{Answer:} $\answer{20}$ percent

\begin{hint}
\textbf{Solution.} The discount is \$20.00, so
$r=\dfrac{20}{99.99}=0.2000\ldots\approx 20\%$.
\end{hint}
\end{problem}

\begin{problem}
\textbf{11.} The sale price of a toboggan is \$77, which is $30\%$ off the regular price.
Find the regular price. \qquad \textbf{Answer:} $\answer{110}$

\begin{hint}
\textbf{Solution.} $77=R-0.3R=0.7R$, so $R=\dfrac{77}{0.7}=\$110$.
\end{hint}
\end{problem}
