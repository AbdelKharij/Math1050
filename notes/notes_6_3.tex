\begin{abstract}
    Section 6.3 --- Solving Systems of Linear Equations by the Addition Method.
    Worked examples, then practice problems with answers.
\end{abstract}

\maketitle

In the addition method, multiply one or both equations by constants so that the
coefficients of one variable are opposites, then add the equations to eliminate that
variable.

\textbf{Note.} A false result such as $0=2$ means \textbf{no solution}. A result that is
always true, such as $0=0$, means the system is \textbf{dependent} and has infinitely
many solutions.

\section*{Worked examples}

\begin{problem}

\textbf{Example 1.} Solve using the addition method.
\[
  \left\{\begin{aligned}
    3x+2y &= 4 \\
    4x-2y &= 10
  \end{aligned}\right.
\]

\begin{hint}
\textbf{Solution.} The $y$ terms are already opposites, so add:
\[
  7x=14 \;\Longrightarrow\; x=2 .
\]
Then $3(2)+2y=4$, so $2y=-2$ and $y=-1$. The solution is $(2,-1)$.
\end{hint}
\end{problem}

\begin{problem}

\textbf{Example 2.} Continuing a system whose elimination gave $x=1$, with
$5x+3y=5$.

\begin{hint}
\textbf{Solution.} Substitute $x=1$:
\[
  5(1)+3y=5 \;\Longrightarrow\; 3y=0 \;\Longrightarrow\; y=0,
\]
so the solution is $(1,0)$.
\end{hint}
\end{problem}

\begin{problem}

\textbf{Example 3.} Solve using the addition method.
\[
  \left\{\begin{aligned}
    -2x-4y &= -22 \\
    x+2y &= 12
  \end{aligned}\right.
\]

\begin{hint}
\textbf{Solution.} Multiply the second equation by 2 and add:
\[
  -2x-4y=-22, \qquad 2x+4y=24 \;\Longrightarrow\; 0=2 .
\]
That is false, so the system has \textbf{no solution}.
\end{hint}
\end{problem}

\begin{problem}

\textbf{Example 4.} Solve using the addition method.
\[
  \left\{\begin{aligned}
    2x+9y &= 5 \\
    5x &= 6-3y
  \end{aligned}\right.
\]

\begin{hint}
\textbf{Solution.} Rewrite the second equation as $5x+3y=6$, multiply it by $-3$, and
add:
\[
  2x+9y=5, \qquad -15x-9y=-18 \;\Longrightarrow\; -13x=-13 \;\Longrightarrow\; x=1 .
\]
Then $2(1)+9y=5$, so $9y=3$ and $y=\tfrac{1}{3}$. The solution is
$\left(1,\tfrac{1}{3}\right)$.
\end{hint}
\end{problem}

\begin{problem}

\textbf{Example 5.} Solve using the addition method.
\[
  \left\{\begin{aligned}
    2x+11y &= 15 \\
    \tfrac{x}{3}+\tfrac{5}{3}y &= 21
  \end{aligned}\right.
\]

\begin{hint}
\textbf{Solution.} Multiply the second equation by 3 to clear fractions, giving
$x+5y=63$; multiply that by $-2$ and add:
\[
  2x+11y=15, \qquad -2x-10y=-126 \;\Longrightarrow\; y=-111 .
\]
Then $2x+11(-111)=15$, so $2x=1236$ and $x=618$. The solution is $(618,-111)$.
\end{hint}
\end{problem}

\begin{problem}

\textbf{Example 6.} Solve using the addition method.
\[
  \left\{\begin{aligned}
    \tfrac{9x}{2}+\tfrac{45y}{2} &= \tfrac{171}{2} \\
    \tfrac{x}{2}+\tfrac{5y}{2} &= \tfrac{19}{2}
  \end{aligned}\right.
\]

\begin{hint}
\textbf{Solution.} Clearing fractions gives $9x+45y=171$ and $x+5y=19$. Multiplying the
second by $-9$ and adding gives $0=0$, so the system is \textbf{dependent} and has
infinitely many solutions.
\end{hint}
\end{problem}

\section*{Your turn}

\begin{problem}

\textbf{1.} Solve using the addition method: $2x+y=3$, $x-y=3$.

\begin{hint}
\textbf{Answer.} $(2,-1)$.
\end{hint}
\end{problem}

\begin{problem}

\textbf{2.} Solve using the addition method: $x-2y=4$, $3x+4y=2$.

\begin{hint}
\textbf{Answer.} $(2,-1)$.
\end{hint}
\end{problem}

\begin{problem}

\textbf{3.} Solve using the addition method: $x+3y=2$, $3x+9y=6$.

\begin{hint}
\textbf{Answer.} The second equation is 3 times the first, so the system is
\textbf{dependent}: infinitely many solutions, every point on $x+3y=2$.
\end{hint}
\end{problem}

\begin{problem}

\textbf{4.} Solve using the addition method: $7x+10y=13$, $4x+5y=6$.

\begin{hint}
\textbf{Answer.} $(-1,2)$.
\end{hint}
\end{problem}

\begin{problem}

\textbf{5.} Solve using the addition method: $3x+5y=-11$, $2x-7y=3$.

\begin{hint}
\textbf{Answer.} $(-2,-1)$. Multiplying by 7 and 5 gives $21x+35y=-77$ and
$10x-35y=15$; adding gives $31x=-62$, so $x=-2$ and then $y=-1$.
\end{hint}
\end{problem}

\begin{problem}

\textbf{6.} Solve using the addition method:
$\dfrac{x}{3}+\dfrac{4}{3}y=\dfrac{16}{3}$, $3x+12y=48$.

\begin{hint}
\textbf{Answer.} Both equations reduce to $x+4y=16$, so the system is \textbf{dependent}:
infinitely many solutions.
\end{hint}
\end{problem}
