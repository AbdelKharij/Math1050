\section*{Perimeter, area and angles}

\begin{problem}
\textbf{21.} Find the complement of a $13^{\circ}$ angle.

\textbf{Answer:} $\answer{77}$

\begin{hint}
\textbf{Solution.} $x+13=90$, so $x=77^{\circ}$.
\end{hint}
\end{problem}

\begin{problem}
\textbf{22.} Find the supplement of a $106^{\circ}$ angle.

\textbf{Answer:} $\answer{74}$

\begin{hint}
\textbf{Solution.} $x+106=180$, so $x=74^{\circ}$.
\end{hint}
\end{problem}

\begin{problem}
\textbf{23.} In the figure below, $AOB$ is a straight angle. Find $m\angle COB$.
\figbox{\begin{tikzpicture}[scale=1.05]
  \draw[bothray] (-2.6,0) -- (2.6,0);
  \node[bdot] at (-2.1,0) {}; \node[ptlbl,below] at (-2.1,-0.05) {A};
  \node[bdot] at (0,0)    {}; \node[ptlbl,below] at (0,-0.05)    {O};
  \node[bdot] at (2.1,0)  {}; \node[ptlbl,below] at (2.1,-0.05)  {B};
  \draw[ray] (0,0) -- (101:2.5);
  \node[bdot] at (101:1.9) {}; \node[ptlbl,left] at (101:1.95) {C};
  \draw[arc] (0.85,0) arc (0:101:0.85);
  \node[lbl] at (50:1.15) {$?$};
  \draw[arc] (180:0.55) arc (180:101:0.55);
  \node[lbl] at (140:0.85) {$79^{\circ}$};
\end{tikzpicture}}

\textbf{Answer:} $\answer{101}$ degrees

\begin{hint}
\textbf{Solution.} $AOB$ is a straight angle, so $m\angle AOC$ and $m\angle COB$ sum to $180^{\circ}$. Then $m\angle COB=180-79=101^{\circ}$.
\end{hint}
\end{problem}

\begin{problem}
\textbf{26.} Find the measure of $\angle A$.
\figbox{\begin{tikzpicture}[scale=1.05]
  \draw[ray] (0,0) -- (0:2.6);
  \draw[ray] (0,0) -- (211:2.6);
  \node[bdot] at (0,0) {};
  \node[ptlbl,below right] at (0.05,-0.12) {A};
  \draw[arc] (0.7,0) arc (0:211:0.7);
  \node[lbl] at (105:1.05) {$211^{\circ}$};
\end{tikzpicture}}

\textbf{Answer:} $\answer{149}$ degrees

\begin{hint}
\textbf{Solution.} The marked angle and $\angle A$ together make a full turn, so $m\angle A=360-211=149^{\circ}$.
\end{hint}
\end{problem}

\begin{problem}
\textbf{27.} Find the perimeter of a triangle with sides 4 ft 5 in., 5 ft 3 in. and 6 ft 11 in.

\begin{hint}
\textbf{Solution.} The feet give 15 ft and the inches give 19 in. $=1$ ft 7 in., so the perimeter is 16 ft 7 in.
\end{hint}
\end{problem}

\begin{problem}
\textbf{28.} Find the perimeter of a rectangle whose dimensions are 5 m by 8 m.
\begin{image}
\includegraphics{10aff6ea-5599-4d61-9876-3e0c025ba08d-6_321_495_409_304}

\begin{problem}
\textbf{30.} Find the circumference of a circle whose radius is 5 in. Use $3.14$ for $\pi$.

\textbf{Answer:} $\answer{31.4}$

\begin{hint}
\textbf{Solution.} $C=2\pi r=2(5)(3.14)=31.4$ in.
\end{hint}
\end{problem}

\begin{problem}
\textbf{31.} Find the area of a rectangle that measures 4 ft by 3 ft.
\begin{image}
\includegraphics{10aff6ea-5599-4d61-9876-3e0c025ba08d-7_250_503_307_259}
\end{image}

\textbf{Answer:} $\answer{12}$

\begin{hint}
\textbf{Solution.} $A=lw=4\cdot 3=12$ ft$^{2}$.
\end{hint}
\end{problem}

\begin{problem}
\textbf{32.} Find the area of a parallelogram whose height is 2 ft and base is 15 ft.
\begin{image}
\includegraphics{10aff6ea-5599-4d61-9876-3e0c025ba08d-7_236_392_813_358}
\end{image}

\textbf{Answer:} $\answer{30}$

\begin{hint}
\textbf{Solution.} $A=bh=15\cdot 2=30$ ft$^{2}$.
\end{hint}
\end{problem}

\begin{problem}
\textbf{33.} Find the area of a triangle whose height is 7 in and base is 3 in.
\begin{image}
\includegraphics{10aff6ea-5599-4d61-9876-3e0c025ba08d-7_301_270_1217_293}
\end{image}

\textbf{Answer:} $\answer{10.5}$

\begin{hint}
\textbf{Solution.} $A=\tfrac{1}{2}bh=\tfrac{1}{2}(3)(7)=10.5$ in$^{2}$.
\end{hint}
\end{problem}

\begin{problem}
\textbf{34.} Find the area of a square whose side is 20 in.
\begin{image}
\includegraphics{10aff6ea-5599-4d61-9876-3e0c025ba08d-7_222_242_1683_216}
\end{image}

\textbf{Answer:} $\answer{400}$

\begin{hint}
\textbf{Solution.} $A=20\cdot 20=400$ in$^{2}$.
\end{hint}
\end{problem}

\begin{problem}
\textbf{35.} Find the area of a circle whose diameter is 3.6 cm. Use $3.14$ for $\pi$.
\begin{image}
\includegraphics{10aff6ea-5599-4d61-9876-3e0c025ba08d-7_307_313_2138_262}
\end{image}

\textbf{Answer:} $\answer{10.174}$

\begin{hint}
\textbf{Solution.} The radius is $1.8$, so $A=\pi r^{2}=(3.14)(1.8)^{2}=10.174$ cm$^{2}$.
\end{hint}
\end{problem}
