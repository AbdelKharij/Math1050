\maketitle

A \textbf{rational number} is a number that can be written as a ratio
$\dfrac{a}{b}$ of two integers with $b \neq 0$. The \textbf{numerator} is the top number
and the \textbf{denominator} is the bottom number.

All three of $\dfrac{-2}{3}$, $\dfrac{2}{-3}$ and $-\dfrac{2}{3}$ name negative two
thirds; a single negative sign may be written in the numerator, in the denominator, or in
front of the fraction.

\section*{Add and subtract rational numbers}

\begin{problem}
\textbf{11.} Simplify: $-\dfrac{1}{4}+\dfrac{3}{4}$ \qquad
\textbf{Answer:} $\answer{\frac{1}{2}}$
\end{problem}

\begin{problem}
\textbf{12.} Simplify: $\dfrac{5}{8}-\left(-\dfrac{3}{8}\right)$ \qquad
\textbf{Answer:} $\answer{1}$
\end{problem}

\begin{problem}
\textbf{13.} Simplify (find the LCM): $\dfrac{2}{3}-\dfrac{1}{2}+\dfrac{5}{6}$

\begin{hint}
\textbf{Answer.} The least common denominator is 6:
$\dfrac{4}{6}-\dfrac{3}{6}+\dfrac{5}{6}=\dfrac{6}{6}=1$.
\end{hint}
\end{problem}

\begin{problem}
\textbf{14.} Simplify: $\dfrac{1}{27}+\dfrac{1}{36}$

\begin{hint}
\textbf{Answer.} The least common denominator is 108:
$\dfrac{4}{108}+\dfrac{3}{108}=\dfrac{7}{108}$.
\end{hint}
\end{problem}

\begin{problem}
\textbf{15.} Simplify: $\dfrac{5}{48}+\dfrac{7}{360}$

\begin{hint}
\textbf{Answer.} The least common denominator is 720:
$\dfrac{75}{720}+\dfrac{14}{720}=\dfrac{89}{720}$.
\end{hint}
\end{problem}

\section*{Convert among percents, fractions and decimals}

\begin{problem}
\textbf{16.} Write as a fraction and as a decimal.

(a) $40\%$ \qquad \textbf{Answer:} $\answer{\frac{2}{5}}$ and $\answer{0.4}$

(b) $175\%$ \qquad \textbf{Answer:} $\answer{\frac{7}{4}}$ and $\answer{1.75}$

(c) $8\%$ \qquad \textbf{Answer:} $\answer{\frac{2}{25}}$ and $\answer{0.08}$
\end{problem}

\begin{problem}
\textbf{17.} Write as a fraction.

(a) $11\frac{1}{9}\%$ \qquad \textbf{Answer:} $\answer{\frac{1}{9}}$

(b) $53\frac{1}{3}\%$ \qquad \textbf{Answer:} $\answer{\frac{8}{15}}$

(c) $99\frac{1}{2}\%$ \qquad \textbf{Answer:} $\answer{\frac{199}{200}}$

\begin{hint}
\textbf{Method.} Write the mixed number as an improper fraction, then multiply by
$\dfrac{1}{100}$. For (a), $11\frac{1}{9}=\dfrac{100}{9}$, so
$\dfrac{100}{9}\cdot\dfrac{1}{100}=\dfrac{1}{9}$.
\end{hint}
\end{problem}

\begin{problem}
\textbf{18.} Write as a percent, rounded to the nearest whole percent:
$1\frac{2}{7}$

\begin{hint}
\textbf{Answer.} $1\frac{2}{7}=\dfrac{9}{7}=1.2857\ldots$, so the percent is
$128.57\ldots\% \approx 129\%$.
\end{hint}
\end{problem}

\textbf{Answer:} $\answer{\frac{3}{5}}$
\end{problem}

\begin{problem}
\textbf{4.} Simplify: $\dfrac{5}{8}+\dfrac{5}{12}+\dfrac{1}{8}$

\begin{hint}
\textbf{Answer.} The least common denominator is 24:
$\dfrac{15}{24}+\dfrac{10}{24}+\dfrac{3}{24}=\dfrac{28}{24}=\dfrac{7}{6}=1\frac{1}{6}$.
\end{hint}
\end{problem}

\begin{problem}
\textbf{5.} Write as a fraction: $19\%$ \qquad \textbf{Answer:} $\answer{\frac{19}{100}}$
\end{problem}

\begin{problem}
\textbf{6.} Write as a fraction: $5\frac{3}{4}\%$

\begin{hint}
\textbf{Answer.} $5\frac{3}{4}=\dfrac{23}{4}$, so the percent is
$\dfrac{23}{4}\cdot\dfrac{1}{100}=\dfrac{23}{400}$.
\end{hint}
\end{problem}
