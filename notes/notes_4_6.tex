    Section 4.6 --- Uniform Motion.
    Worked examples, then practice problems with answers.

\maketitle

\textbf{Uniform motion equation:} $rt=d$, where $r$ is the rate, $t$ is the time and $d$
is the distance.

Set up a rate--time--distance table for each moving object, then use whatever the problem
tells you is equal: the two distances, their sum, or the total time.

\section*{Worked examples}

\begin{problem}
\textbf{1.} A car leaves town traveling at 35 mph. Two hours later, a second car leaves
the same town, on the same road, traveling at 55 mph. How many hours after the second car
leaves will the second car pass the first car?

\begin{hint}
\textbf{Solution.} The distances are equal, and the first car has been driving $t+2$
hours:
\[
  35(t+2)=55t \;\Longrightarrow\; 35t+70=55t \;\Longrightarrow\; 20t=70
  \;\Longrightarrow\; t=3.5 \text{ hours}.
\]
\end{hint}
\end{problem}

\begin{problem}
\textbf{2.} Two cars, the first traveling 10 mph faster than the second, start at the same
time from the same point and travel in opposite directions. In 3 hours they are 288 miles
apart. Find the rate of the second car.

\begin{hint}
\textbf{Solution.} Travelling apart, the distances add. With $r$ the second car's rate,
\[
  3(r+10)+3r=288 \;\Longrightarrow\; 6r+30=288 \;\Longrightarrow\; r=43 \text{ mph}.
\]
\end{hint}
\end{problem}

\begin{problem}
\textbf{3.} A bicycling club rides out into the country at 16 mph and returns over the
same road at 12 mph. How far does the club ride out into the country if it travels for a
total of 7 hours?

\begin{hint}
\textbf{Solution.} The distance out equals the distance back, and the times add to 7:
\[
  \frac{d}{16}+\frac{d}{12}=7 \;\Longrightarrow\; \frac{7d}{48}=7
  \;\Longrightarrow\; d=48 \text{ miles}.
\]
\end{hint}
\end{problem}

\begin{problem}
\textbf{4.} An executive drove from home at an average speed of 30 mph to an airport where
a helicopter was waiting, then flew to the corporate offices at an average speed of
60 mph. The entire distance was 150 mi and the trip took 3 hours. Find the distance from
the airport to the corporate offices.

\begin{hint}
\textbf{Solution.} With $t$ the driving time, the flying time is $3-t$:
\[
  30t+60(3-t)=150 \;\Longrightarrow\; 180-30t=150 \;\Longrightarrow\; t=1 .
\]
So the helicopter flew for 2 hours, covering $60(2)=120$ miles.
\end{hint}
\end{problem}

\begin{problem}
\textbf{5.} At noon a train leaves Washington DC headed for Charleston, South Carolina, a
distance of 500 mi, travelling at 60 mph. At 1 PM a second train leaves Charleston headed
for Washington DC, travelling at 50 mph. How long after the train leaves Charleston will
the two trains pass each other?

\begin{hint}
\textbf{Solution.} By 1 PM the first train has covered $60$ mi, leaving $440$ mi between
them, and they close at $60+50=110$ mph:
\[
  110t=440 \;\Longrightarrow\; t=4 \text{ hours}.
\]
\end{hint}

\begin{problem}
\textbf{1.} Two planes start from the same point and fly in opposite directions. The first
plane is flying 25 mph slower than the second. In 2 hours the planes are 470 miles apart.
Find the rate of each plane.

\begin{hint}
\textbf{Answer.} Flying apart, the distances add:
\[
  2r+2(r+25)=470 \;\Longrightarrow\; 4r+50=470 \;\Longrightarrow\; r=105 .
\]
The planes fly at $105$ mph and $130$ mph.

\end{hint}
\end{problem}

\begin{problem}
\textbf{2.} Michael leaves a dock in his motorboat and travels at an average speed of
9 mph toward the Isle of Shoals. Two hours later a tour boat leaves the same dock and
travels at an average speed of 18 mph toward the same island. How many hours after the
tour boat leaves will Michael's boat be alongside the tour boat?

\begin{hint}
\textbf{Answer.} The distances are equal, and Michael has been travelling $t+2$ hours:
\[
  9(t+2)=18t \;\Longrightarrow\; 18=9t \;\Longrightarrow\; t=2 \text{ hours}.
\]

\begin{problem}
\textbf{3.} After a sailboat had been on the water for 3 hours, a change in the wind
reduced its average speed by 5 mph. The entire distance sailed was 57 miles and the total
time was 6 hours. How far did the sailboat travel in the first 3 hours?

\begin{hint}
\textbf{Answer.} The two legs are 3 hours each, at $r$ and $r-5$:
\[
  3r+3(r-5)=57 \;\Longrightarrow\; 6r-15=57 \;\Longrightarrow\; r=12 .
\]
The first 3 hours covered $12(3)=36$ miles.

\end{image}
\end{hint}
\end{problem}

\begin{problem}
\textbf{4.} A passenger train leaves a depot 2 hours after a freight train leaves the same
depot. The freight train travels 20 mph slower than the passenger train. Find the rate of
each train if the passenger train overtakes the freight train in 3 hours.

\begin{hint}
\textbf{Answer.} When the passenger train catches up it has travelled 3 hours and the
freight train 5 hours, over equal distances. With $p$ the passenger rate,
\[
  3p=5(p-20) \;\Longrightarrow\; 3p=5p-100 \;\Longrightarrow\; p=50 .
\]
The passenger train travels $50$ mph and the freight train $30$ mph.

\end{image}
\end{hint}
\end{problem}

\begin{problem}
\textbf{5.} A race car driver starts along a 50-mile race course travelling at an average
speed of 90 mph. Fifteen minutes later a second driver starts along the same course at an
average speed of 120 mph. Will the second car overtake the first car before the drivers
reach the end of the course?

\begin{hint}
\textbf{Answer.} The first car has a $\tfrac{1}{4}$-hour head start, so
\[
  90\left(t+\tfrac{1}{4}\right)=120t \;\Longrightarrow\; 30t=22.5
  \;\Longrightarrow\; t=\tfrac{3}{4}\text{ hour}.
\]
That would happen at $120\left(\tfrac{3}{4}\right)=90$ miles, well past the end of the
50-mile course, so \textbf{no} --- the second car does not overtake the first before the
finish.
\end{hint}
\end{problem}
