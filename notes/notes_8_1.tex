\begin{abstract}
    Section 8.1 --- Common Factors and Factoring by Grouping.
    Practice problems with answers.
\end{abstract}

\maketitle

To factor a monomial from a polynomial, find the greatest common factor (GCF) of the
terms and divide it out. To factor by grouping, look for a common binomial factor,
remembering that $(b-a) = -(a-b)$.

\section*{Greatest common factor}

\begin{problem}

\textbf{1.} Find the GCF.

(a) $12a^{4}b$ and $18a^{2}b^{2}c$ \qquad

\begin{hint}
\textbf{Answer.} $6a^{2}b$.

(b) $63y$, $189y$ and $135y$ \qquad \textbf{Answer.} $9y$.

(c) $40$, $120$ and $119$ \qquad \textbf{Answer.} $1$ (119 is $7 \cdot 17$, sharing no
factor with 40).
\end{hint}
\end{problem}

\begin{problem}

\textbf{2.} Find the GCF of $14a^{3}$ and $49a^{7}$. \qquad

\begin{hint}
\textbf{Answer.} $7a^{3}$.
\end{hint}
\end{problem}

\begin{problem}

\textbf{3.} Find the GCF of $15a^{4}b^{2}$, $9ab^{5}$ and $18ab^{2}$. \qquad

\begin{hint}
\textbf{Answer.} $3ab^{2}$.
\end{hint}
\end{problem}

\section*{Factor out the common factor}

\begin{problem}

\textbf{4.} Factor.

(a) $5a+5$ \qquad

\begin{hint}
\textbf{Answer.} $5(a+1)$.

(b) $16-8a^{2}$ \qquad \textbf{Answer.} $8\left(2-a^{2}\right)$.

(c) $30x-6$ \qquad \textbf{Answer.} $6(5x-1)$.

(d) $6a^{2}bc+4ab^{2}c$ \qquad \textbf{Answer.} $2abc(3a+2b)$.

(e) $a^{3}+4a^{2}+8a$ \qquad \textbf{Answer.} $a\left(a^{2}+4a+8\right)$.

(f) $2x^{3}-12x+8$ \qquad \textbf{Answer.} $2\left(x^{3}-6x+4\right)$.

(g) $-55ab+65a^{2}b$ \qquad \textbf{Answer.} $5ab(13a-11)$.

(h) $63x^{2}y+77xy-42xy^{2}$ \qquad \textbf{Answer.} $7xy(9x+11-6y)$.
\end{hint}
\end{problem}

\begin{problem}

\textbf{5.} Factor: $12+12y^{2}$. \qquad

\begin{hint}
\textbf{Answer.}
$12\left(1+y^{2}\right)$.
\end{hint}
\end{problem}

\begin{problem}

\textbf{6.} Factor: $3a^{2}+5a^{5}$. \qquad

\begin{hint}
\textbf{Answer.}
$a^{2}\left(3+5a^{3}\right)$.
\end{hint}
\end{problem}

\begin{problem}

\textbf{7.} Factor: $8x^{2}y^{3}-4x^{2}$. \qquad

\begin{hint}
\textbf{Answer.}
$4x^{2}\left(2y^{3}-1\right)$.
\end{hint}
\end{problem}

\begin{problem}

\textbf{8.} Factor: $x^{3}-3x^{2}-x$. \qquad

\begin{hint}
\textbf{Answer.}
$x\left(x^{2}-3x-1\right)$.
\end{hint}
\end{problem}

\section*{Common binomial factors}

\begin{problem}

\textbf{9.} Factor.

(a) $x(3+5)+y(3+5)$ \qquad

\begin{hint}
\textbf{Answer.} $8(x+y)$.

(b) $5(x+y)+3(x+y)$ \qquad \textbf{Answer.} $8(x+y)$.

(c) $y(x+2)+3(x+2)$ \qquad \textbf{Answer.} $(x+2)(y+3)$.

(d) $a(b-7)+b(b-7)$ \qquad \textbf{Answer.} $(b-7)(a+b)$.
\end{hint}
\end{problem}

\begin{problem}

\textbf{10.} Factor out a $(-1)$.

(a) $(x-2)$ \qquad

\begin{hint}
\textbf{Answer.} $-(2-x)$.

(b) $(2-x)$ \qquad \textbf{Answer.} $-(x-2)$.

(c) $(a-b)$ \qquad \textbf{Answer.} $-(b-a)$.

(d) $(b-a)$ \qquad \textbf{Answer.} $-(a-b)$.
\end{hint}
\end{problem}

\begin{problem}

\textbf{11.} Factor.

(a) $a(x-2)-b(2-x)$ \qquad

\begin{hint}
\textbf{Answer.} $(x-2)(a+b)$, since $-(2-x)=(x-2)$.

(b) $b(a-7)+3(7-a)$ \qquad \textbf{Answer.} $(a-7)(b-3)$.
\end{hint}
\end{problem}

\begin{problem}

\textbf{12.} Factor: $x(b+7)-y(b+7)$. \qquad

\begin{hint}
\textbf{Answer.} $(b+7)(x-y)$.
\end{hint}
\end{problem}

\begin{problem}

\textbf{13.} Factor: $c(x-6)-7(x-6)$. \qquad

\begin{hint}
\textbf{Answer.} $(x-6)(c-7)$.
\end{hint}
\end{problem}

\begin{problem}

\textbf{14.} Factor: $y(ab+4)-7(ab+4)$. \qquad

\begin{hint}
\textbf{Answer.} $(ab+4)(y-7)$.
\end{hint}
\end{problem}

\begin{problem}

\textbf{15.} Factor: $x(a-2b)-y(a-2b)$. \qquad

\begin{hint}
\textbf{Answer.} $(a-2b)(x-y)$.
\end{hint}
\end{problem}

\begin{problem}

\textbf{16.} Factor: $c(a-2)-b(2-a)$. \qquad

\begin{hint}
\textbf{Answer.} $(a-2)(b+c)$.
\end{hint}
\end{problem}

\begin{problem}

\textbf{17.} Factor: $w(3x-4)-(4-3x)$.

\begin{hint}
\textbf{Answer.} Rewrite $-(4-3x)$ as $+1(3x-4)$:
$w(3x-4)+1(3x-4)=(3x-4)(w+1)$.
\end{hint}
\end{problem}

\begin{problem}

\textbf{18.} Factor: $a(x-y)-2(y-x)$. \qquad

\begin{hint}
\textbf{Answer.} $(x-y)(a+2)$.
\end{hint}
\end{problem}

\begin{problem}

\textbf{19.} Factor: $z(c+5)-8(5+c)$. \qquad

\begin{hint}
\textbf{Answer.} $(c+5)(z-8)$.
\end{hint}
\end{problem}

\section*{Factor by grouping}

\begin{problem}

\textbf{20.} Factor.

(a) $xy+3y+2x+6$ \qquad

\begin{hint}
\textbf{Answer.} $(x+3)(y+2)$.

(b) $a^{3}+8a-a^{2}-8$ \qquad \textbf{Answer.} $(a-1)\left(a^{2}+8\right)$.

(c) $3y^{3}-12y^{2}+y-4$ \qquad \textbf{Answer.} $(y-4)\left(3y^{2}+1\right)$.

(d) $x^{2}-3x+4ax-12a$ \qquad \textbf{Answer.} $(x-3)(x+4a)$.

(e) $ab+3b-2a-6$ \qquad \textbf{Answer.} $(a+3)(b-2)$.

(f) $3ax-3bx-2ay+2by$ \qquad \textbf{Answer.} $(a-b)(3x-2y)$.
\end{hint}
\end{problem}

\begin{problem}

\textbf{21.} Factor: $x^{2}y+4x^{2}+3y+12$. \qquad

\begin{hint}
\textbf{Answer.}
$\left(x^{2}+3\right)(y+4)$.
\end{hint}
\end{problem}

\begin{problem}

\textbf{22.} Factor: $x^{3}-4x^{2}-3x+12$. \qquad

\begin{hint}
\textbf{Answer.}
$(x-4)\left(x^{2}-3\right)$.
\end{hint}
\end{problem}
