\maketitle

Multiply using the area model or FOIL. Two patterns worth memorising:
\[
  (x+c)^{2}=x^{2}+2cx+c^{2}, \qquad (x+c)(x-c)=x^{2}-c^{2}.
\]

\section*{Multiply using the area model}


\begin{problem}

\textbf{1.} $-3x\left(x^{2}-4x+2\right)$

\begin{hint}
\textbf{Answer.} $- 3 x^{3} + 12 x^{2} - 6 x$.
\end{hint}
\end{problem}

\begin{problem}

\textbf{2.} $(3x+7)(5x+2)$

\begin{hint}
\textbf{Answer.} $15 x^{2} + 41 x + 14$.
\end{hint}
\end{problem}

\begin{problem}

\textbf{3.} $(2x+4)(5x+9)$

\begin{hint}
\textbf{Answer.} $10 x^{2} + 38 x + 36$.
\end{hint}
\end{problem}

\begin{problem}

\textbf{4.} $(2x+3)^{2}$

\begin{hint}
\textbf{Answer.} $4 x^{2} + 12 x + 9$.
\end{hint}
\end{problem}

\begin{problem}

\textbf{5.} $(4x-3)(4x+3)$

\begin{hint}
\textbf{Answer.} $16 x^{2} - 9$.
\end{hint}
\end{problem}

\begin{problem}

\textbf{6.} $\left(\frac{2}{3}x+5\right)\left(\frac{2}{3}x-5\right)$

\begin{hint}
\textbf{Answer.} $\frac{4 x^{2}}{9} - 25$.
\end{hint}
\end{problem}

\begin{problem}

\textbf{7.} The radius of a circle is $(x-4)$ ft. Find the area of the circle, leaving the answer in terms of $\pi$.

\begin{hint}
\textbf{Answer.} $\pi x^{2} - 8 \pi x + 16 \pi$.
\end{hint}
\end{problem}

\begin{problem}

\textbf{8.} Find the area of a square piece of land whose side is $\left(\frac{8}{9}x-15\right)$ km.

\begin{hint}
\textbf{Answer.} $\frac{64 x^{2}}{81} - \frac{80 x}{3} + 225$.
\end{hint}
\end{problem}

\begin{problem}

\textbf{9.} Find the area of a triangular piece of land with base $(8t+8)$ km and height $(8t-8)$ km.

\begin{hint}
\textbf{Answer.} $32 t^{2} - 32$.
\end{hint}
\end{problem}

\begin{problem}

\textbf{10.} A softball diamond has dimensions 45 ft by 45 ft. A base path border $x$ feet
wide lies on both the first-base side and the third-base side. Express the total area of
the diamond and the base path in terms of $x$.

\begin{hint}
\textbf{Answer.} $x^{2} + 90 x + 2025$.

\end{image}
\end{hint}
\end{problem}
