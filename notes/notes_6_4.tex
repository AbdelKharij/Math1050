\maketitle

For rate-of-wind and rate-of-current problems, use $r\cdot t = d$ (rate $\times$ time $=$
distance). If $r$ is the rate in still air or calm water and $w$ is the rate of the wind
or current, then the rate with the wind is $r+w$ and against it is $r-w$.

\section*{Worked examples}

\begin{problem}
\textbf{Example 1.} A 600-mile trip takes 4 hours flying with the wind, and the return
trip against the wind takes 5 hours. Find the rate of the plane in still air and the rate
of the wind.

\textbf{Solution.}
\begin{center}
\begin{tabular}{|l|l|l|l|}
\hline
 & Rate & Time & Distance \\
\hline
With the wind & $r+w$ & 4 & 600 \\
\hline
Against the wind & $r-w$ & 5 & 600 \\
\hline
\end{tabular}
\end{center}
From $4(r+w)=600$ and $5(r-w)=600$ we get $r+w=150$ and $r-w=120$. Adding gives
$2r=270$, so $r=135$ mph and $w=15$ mph.
\end{problem}

\begin{problem}
\textbf{Example 2.} The sum of two numbers is 227 and their difference is 3. Find the
product of the two numbers.

\textbf{Solution.} From $x+y=227$ and $x-y=3$, adding gives $2x=230$, so $x=115$ and
$y=112$. The product is $115 \cdot 112 = 12{,}880$.
\end{problem}

\begin{problem}
\textbf{Example 3.} Tickets to a community production are priced so that 2 regular
tickets with 3 student tickets cost \$32.75, and 7 regular tickets with 1 student ticket
cost \$64.75. How much would 5 regular tickets and 7 student tickets cost?

\textbf{Solution.} With $r$ the regular price and $s$ the student price,
\[
  2r+3s=32.75, \qquad 7r+s=64.75 .
\]
Multiplying the second by $-3$ and adding gives $-19r=-161.50$, so $r=\$8.50$, and then
$3s=15.75$ gives $s=\$5.25$. Therefore
$5(8.50)+7(5.25)=\$79.25$.
\end{problem}

\begin{problem}
\textbf{Example 4.} Joe buys 8 gallons of gas and 2 bottles of super-soda for \$21.20.
Rick spends \$20.10 on 6 gallons of gas and 5 bottles of super-soda. What is the price of
a gallon of gas?

\textbf{Solution.} With $g$ the price per gallon and $b$ the price per bottle,
\[
  8g+2b=21.20, \qquad 6g+5b=20.10 .
\]
Multiplying the first by 5 and the second by $-2$ gives $40g+10b=106$ and
$-12g-10b=-40.20$; adding gives $28g=65.80$, so $g=\$2.35$ per gallon.
\end{problem}

\begin{problem}
\textbf{Example 5.} A plane flying with the jet stream flew 2250 miles in 5 hours.
Flying against it, the plane flew only 1750 miles in the same time. Find the rate of the
plane in calm air and the rate of the wind.

\textbf{Solution.} $r+w=\tfrac{2250}{5}=450$ and $r-w=\tfrac{1750}{5}=350$, so $r=400$
mph and $w=50$ mph.
\end{problem}

\begin{problem}
\textbf{Example 6.} A merchant mixes 4 lb of cinnamon tea with 1 lb of spice tea to make
a mixture costing \$12 per pound. Mixing 1 lb of cinnamon tea with 4 lb of spice tea
costs \$15 per pound. Find the cost per pound of each tea.

\textbf{Solution.} With $c$ and $s$ the costs per pound, the 5-pound mixtures give
\[
  4c+s=5(12)=60, \qquad c+4s=5(15)=75 .
\]
Solving gives $c=\$11$ per pound for cinnamon and $s=\$16$ per pound for spice.
\end{problem}

\section*{Your turn}

\begin{problem}
\textbf{1.} You have 33 coins (all dimes and quarters) worth \$4.95. How many dimes do
you have?

\textbf{Answer.} 22 dimes (and 11 quarters), from $d+q=33$ and $10d+25q=495$.
\end{problem}

\begin{problem}
\textbf{2.} Flying with a tailwind a jet can fly 2460 miles in 4 hours. Flying against
the same wind, it covers 1780 miles in the same time. Determine the speed of the jet in
calm air.

\textbf{Answer.} $r+w=615$ and $r-w=445$, so the jet flies $530$ mph in calm air (and the
wind is $85$ mph).
\end{problem}

\begin{problem}
\textbf{3.} Flying with a tailwind a jet can fly 1350 miles in 2 hours. Flying against
the same wind, it takes 3 hours to cover 1455 miles. Determine the wind speed.

\textbf{Answer.} $r+w=675$ and $r-w=485$, so the wind is $95$ mph (and the jet flies
$580$ mph in calm air).
\end{problem}

\begin{problem}
\textbf{4.} A whale swimming against an ocean current traveled 60 miles in 2 hours.
Swimming with the current, it travelled the same distance in 1.5 hours. Find the speed of
the whale in calm water and the rate of the current.

\textbf{Answer.} $r-c=30$ and $r+c=40$, so the whale swims $35$ mph in calm water and the
current is $5$ mph.
\end{problem}

\begin{problem}
\textbf{5.} A motorboat traveling with the current went 35 miles in 3.5 hours. Traveling
against the current, it went 12 miles in 3 hours. Find the rate of the boat in calm water
and the rate of the current.

\textbf{Answer.} $r+c=10$ and $r-c=4$, so the boat travels $7$ mph in calm water and the
current is $3$ mph.
\end{problem}
