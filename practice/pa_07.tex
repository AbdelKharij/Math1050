\begin{problem}
\textbf{1.} Simplify: $4x-2(3x+8)$.

\textbf{Solution.} Distribute the $-2$, then combine like terms:
\[
  4x-2(3x+8) = 4x-6x-16 = -2x-16 .
\]
\end{problem}

\begin{problem}
\textbf{2.} Solve: $2x+3x=20$.

\textbf{Solution.}
\[
  5x=20 \;\Longrightarrow\; x=4 .
\]
\end{problem}
