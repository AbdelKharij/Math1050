\begin{enumerate}
  \item Solve for $x$ :
\end{enumerate}

$$\begin{aligned}
& 5 x+3=13 \\
& 5 x+3-3=13-3 \\
& 5 x=10 \\
& \frac{5 x}{5} x=2
\end{aligned}$$

\begin{enumerate}
  \setcounter{enumi}{1}
  \item Solve for $x$ : $\frac{3 x}{2}+1=4$
\end{enumerate}

$$\begin{aligned}
& \frac{3 x}{2}+1-1=4-1 \\
& \frac{3 x}{2}=3 \\
& \frac{3 x}{2} \cdot 2=3 \cdot 2 \\
& 3 x=6 \\
& \frac{3 x}{3}=\frac{6}{3} \\
& x=2
\end{aligned}$$
