\begin{problem}

\textbf{1.} Solve for $x$: $5x+3=13$.

\begin{hint}
\textbf{Solution.}
\[
  5x+3-3=13-3 \;\Longrightarrow\; 5x=10 \;\Longrightarrow\; x=2 .
\]
\end{hint}
\end{problem}

\begin{problem}

\textbf{2.} Solve for $x$: $\dfrac{3x}{2}+1=4$.

\begin{hint}
\textbf{Solution.}
\[
  \frac{3x}{2}=3 \;\Longrightarrow\; 3x=6 \;\Longrightarrow\; x=2 .
\]
\end{hint}
\end{problem}
