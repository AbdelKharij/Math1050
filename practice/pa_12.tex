\begin{problem}

\textbf{1.} (2 pts) In the figure below, lines $f$ and $g$ are parallel. Give the degree
measure of the following angles.
\begin{image}
\includegraphics{b1aad74d-5334-4184-b6dd-2061adef7fc3-1_594_1092_406_277}
\end{image}

\begin{hint}
\textbf{Solution.} Angle $a=120^{\circ}$ and angle $b=120^{\circ}$, since angles formed
by a transversal cutting parallel lines are either equal or supplementary.
\end{hint}
\end{problem}

\begin{problem}

\textbf{2.} (2 pts) In the figure below, the angles of a triangle are $x$, $2x$ and $3x$.
Find the value of $x$.
\begin{image}
\includegraphics{b1aad74d-5334-4184-b6dd-2061adef7fc3-1_534_432_1669_335}
\end{image}

\begin{hint}
\textbf{Solution.} The angles of a triangle sum to $180^{\circ}$:
\[
  x+2x+3x=180 \;\Longrightarrow\; 6x=180 \;\Longrightarrow\; x=30^{\circ}.
\]
\end{hint}
\end{problem}
