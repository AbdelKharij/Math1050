\section*{Multiple choice}

For problems 1--4, choose the one correct answer.

\begin{problem}

\textbf{1.} Find the complement of a $24^{\circ}$ angle.

\begin{multipleChoice}
\choice{$48^{\circ}$}
\choice{$336^{\circ}$}
\choice[correct]{$66^{\circ}$}
\choice{$69^{\circ}$}
\choice{$156^{\circ}$}
\end{multipleChoice}

\begin{hint}
\textbf{Solution.} Complementary angles sum to $90^{\circ}$, so $90-24=66^{\circ}$.
\end{hint}
\end{problem}

\begin{problem}

\textbf{2.} 12 is what percent of 80?

\begin{multipleChoice}
\choice{$60\%$}
\choice{$12\%$}
\choice[correct]{$15\%$}
\choice{$85\%$}
\choice{$80\%$}
\end{multipleChoice}

\begin{hint}
\textbf{Solution.} $\dfrac{12}{80}=0.15=15\%$.
\end{hint}
\end{problem}

\begin{problem}

\textbf{3.} The temperature at which radon boils is $-62^{\circ}\mathrm{C}$. Radon freezes at
$-71^{\circ}\mathrm{C}$. Find the difference between the temperature at which radon boils and
the temperature at which it freezes.

\begin{multipleChoice}
\choice{$-9^{\circ}\mathrm{C}$}
\choice{$100^{\circ}\mathrm{C}$}
\choice[correct]{$9^{\circ}\mathrm{C}$}
\choice{$5^{\circ}\mathrm{C}$}
\choice{$-133^{\circ}\mathrm{C}$}
\end{multipleChoice}

\begin{hint}
\textbf{Solution.} $-62-(-71)=-62+71=9^{\circ}\mathrm{C}$.
\end{hint}
\end{problem}

\begin{problem}

\textbf{4.} Evaluate the variable expression $c^{2}-3(b\div c)$ when $b=-8$ and $c=-4$.

\begin{multipleChoice}
\choice[correct]{$10$}
\choice{$-18$}
\choice{$21$}
\choice{$-14$}
\choice{$5$}
\end{multipleChoice}

\begin{hint}
\textbf{Solution.} $(-4)^{2}-3\bigl((-8)\div(-4)\bigr)=16-3(2)=16-6=10$.
\end{hint}
\end{problem}

\begin{problem}

\textbf{5.} Evaluate using the Order of Operations Agreement: $-13^{2}+5(4\div 2)$.

\textbf{Answer:} $\answer{-159}$

\begin{hint}
\textbf{Solution.} The exponent applies only to $13$, so $-13^{2}=-169$. Then
\[
  -169+5(4\div 2)=-169+5(2)=-169+10=-159 .
\]
\end{hint}
\end{problem}

\begin{problem}

\textbf{6.} Simplify: $-12\div 0$.

\begin{hint}
\textbf{Solution.} Division by zero is undefined, so this expression has no value.
\end{hint}
\end{problem}

\begin{problem}

\textbf{7.} Simplify: $7-(-14)-5$.

\textbf{Answer:} $\answer{16}$

\begin{hint}
\textbf{Solution.} $7+14-5=21-5=16$.
\end{hint}
\end{problem}

\begin{problem}

\textbf{8.} Simplify: $\left(\dfrac{8}{15}\right)\left(-\dfrac{13}{16}\right)\left(-\dfrac{7}{11}\right)$

\textbf{Answer:} $\answer{91/330}$

\begin{hint}
\textbf{Solution.} Two negative factors give a positive product:
\[
  \frac{8\cdot 13\cdot 7}{15\cdot 16\cdot 11}=\frac{728}{2640}=\frac{91}{330}.
\]
\end{hint}
\end{problem}

\begin{problem}

\textbf{10.} Compute the area of a triangle whose height is 10 in.\ and base is 4 in.

\begin{image}

\textbf{Solution.} $A=\tfrac{1}{2}bh=\tfrac{1}{2}(4)(10)=20$ in$^{2}$.
\end{hint}

\begin{problem}

\textbf{11.} Translate into a variable expression: ``the product of eight and the total of
$n$ and two''.

\textbf{Answer:} $\answer{8(n+2)}$

\begin{hint}
\textbf{Solution.} ``The total of $n$ and two'' is $n+2$, and the product of eight and that
total is $8(n+2)$.
\end{hint}
\end{problem}

\begin{problem}

\textbf{12.} Simplify the expression: $\left(\dfrac{7}{9}\right)\div\left(-\dfrac{1}{12}\right)$

\textbf{Answer:} $\answer{-28/3}$

\begin{hint}
\textbf{Solution.} Multiply by the reciprocal:
$\dfrac{7}{9}\cdot(-12)=-\dfrac{84}{9}=-\dfrac{28}{3}$.
\end{hint}
\end{problem}

\begin{problem}

\textbf{13.} Solve for $x$: $-5x=10$

\textbf{Answer:} $x=\answer{-2}$

\begin{hint}
\textbf{Solution.} Divide both sides by $-5$: $x=\dfrac{10}{-5}=-2$.
\end{hint}
\end{problem}

\begin{problem}

\textbf{14.} Solve for $x$: $\dfrac{2}{3}x-1=\dfrac{1}{3}x+2$

\textbf{Answer:} $x=\answer{9}$

\begin{hint}
\textbf{Solution.} Subtract $\tfrac{1}{3}x$ from both sides to get $\tfrac{1}{3}x-1=2$, so
$\tfrac{1}{3}x=3$ and $x=9$.
\end{hint}
\end{problem}

\begin{problem}

\textbf{15.} Solve for $x$: $1-8x=28-5x$

\textbf{Answer:} $x=\answer{-9}$

\begin{hint}
\textbf{Solution.} Add $5x$ to both sides: $1-3x=28$. Then $-3x=27$, so $x=-9$.
\end{hint}
\end{problem}

\begin{problem}

\textbf{16.} Simplify: $7(x+8)-5(x-6)$

\textbf{Answer:} $\answer{2x+86}$

\begin{hint}
\textbf{Solution.} $7x+56-5x+30=2x+86$.
\end{hint}
\end{problem}

\begin{problem}

\textbf{17.} Find the circumference of a circle whose radius is 18.4 cm. Use $3.14$ for
$\pi$ and write all the digits of your answer.

\textbf{Answer:} $\answer{115.552}$ cm

\begin{hint}
\textbf{Solution.} $C=2\pi r=2(3.14)(18.4)=115.552$ cm.
\end{hint}
\end{problem}

\begin{problem}

\textbf{18.} At 5:00 am a train leaves a station and travels at a rate of 50 mph. At 6:00 am
another train leaves the same station on the same track and travels in the direction of the
first train at a speed of 70 mph. At 7:00 am, how many miles separate the two trains?

\textbf{Answer:} $\answer{30}$ miles

\begin{hint}
\textbf{Solution.} By 7:00 am the first train has travelled for 2 hours,
$d=50(2)=100$ miles. The second train has travelled for 1 hour, $d=70(1)=70$ miles.
They are $100-70=30$ miles apart.
\end{hint}
\end{problem}

\begin{problem}

\textbf{19.} $40\%$ of what number is 8?

\textbf{Answer:} $\answer{20}$

\begin{hint}
\textbf{Solution.} $0.40x=8$, so $x=\dfrac{8}{0.40}=20$.
\end{hint}
\end{problem}

\begin{problem}

\textbf{20.} Solve for $x$: $\dfrac{3x-2}{8}=2$

\textbf{Answer:} $x=\answer{6}$

\begin{hint}
\textbf{Solution.} Multiply both sides by 8: $3x-2=16$. Then $3x=18$, so $x=6$.
\end{hint}
\end{problem}

\begin{problem}

\textbf{21.} In the figure below, $m\angle ABC=120^{\circ}$ and $m\angle ABD=54^{\circ}$.
Find $m\angle DBC$.

\begin{image}
\includegraphics{26a6a839-371f-4814-8708-5ebdb60d6966-5_324_520_2155_199}
\end{image}

\textbf{Answer:} $\answer{66}$ degrees

\begin{hint}
\textbf{Solution.} $\angle ABD$ sits inside $\angle ABC$, so
$m\angle DBC=120-54=66^{\circ}$.
\end{hint}
\end{problem}

\begin{problem}

\textbf{22.} Find the perimeter of a triangle with sides 3 ft 5 in., 5 ft 8 in.\ and
4 ft 10 in.

\textbf{Answer:} $\answer{13}$ ft $\answer{11}$ in.

\begin{hint}
\textbf{Solution.} The feet add to $3+5+4=12$ ft and the inches add to $5+8+10=23$ in.
Since $23$ in.\ $=1$ ft 11 in., the perimeter is $13$ ft 11 in.
\end{hint}
\end{problem}

\begin{problem}

\textbf{23.} Solve for $x$: $\dfrac{x}{5}+1=6$

\textbf{Answer:} $x=\answer{25}$

\begin{hint}
\textbf{Solution.} Subtract 1 to get $\dfrac{x}{5}=5$, then multiply by 5 to get $x=25$.
\end{hint}
\end{problem}

\begin{problem}

\textbf{24.} Find $25\%$ of 16.

\textbf{Answer:} $\answer{4}$

\begin{hint}
\textbf{Solution.} $0.25(16)=4$.
\end{hint}
\end{problem}

\begin{problem}

\textbf{25.} Simplify. Show all your work:
$\dfrac{6}{11}+\dfrac{9}{22}+\dfrac{11}{24}$

\textbf{Answer:} $\answer{373/264}$

\begin{hint}
\textbf{Solution.} The least common denominator of 11, 22 and 24 is 264:
\[
  \frac{6}{11}\cdot\frac{24}{24}=\frac{144}{264},\qquad
  \frac{9}{22}\cdot\frac{12}{12}=\frac{108}{264},\qquad
  \frac{11}{24}\cdot\frac{11}{11}=\frac{121}{264}.
\]
Adding gives $\dfrac{144+108+121}{264}=\dfrac{373}{264}$.
\end{hint}
\end{problem}
