\section*{Multiple choice (1--5)}

For problems 1--5, choose the one correct answer. Only one
answer is correct.

\begin{problem}

\textbf{1.} Find the equation of the line with $y$-intercept at $(0,-1)$ and slope
$-\frac{5}{21}$.



A) $y=\frac{5}{21}x+1$

B) $y=\frac{5}{21}x-1$

C) $y=-\frac{5}{21}x+2$

D) $y=-\frac{5}{21}x+1$

E) $y=-\frac{5}{21}x-1$

\begin{hint}
\textbf{Solution.} Slope-intercept form is $y=mx+b$ with $m=-\frac{5}{21}$ and $b=-1$:
\[
  y=-\frac{5}{21}x-1 .
\]
That is choice \textbf{E}.
\end{hint}
\end{problem}

\begin{problem}

\textbf{2.} Divide:
\[
  \frac{2x^{10}-4x^{7}-2x^{5}}{x^{2}}
\]



A) $2x^{8}-4x^{5}-2x^{3}$

D) $2x^{10}-4x^{5}-2x^{5}$

E) $2x^{10}-4x^{7}-2x^{3}$

\begin{hint}
\textbf{Solution.} Divide each term by $x^{2}$:
\[
  \frac{2x^{10}}{x^{2}}-\frac{4x^{7}}{x^{2}}-\frac{2x^{5}}{x^{2}}
  = 2x^{8}-4x^{5}-2x^{3}.
\]
That is choice \textbf{A}.
\end{hint}
\end{problem}

\begin{problem}

\textbf{3.} Evaluate: $5^{-2}$.



A) $\frac{1}{25}$

B) $10$

C) $-10$

D) $25$

E) $\frac{1}{10}$

\begin{hint}
\textbf{Solution.} $5^{-2}=\dfrac{1}{5^{2}}=\dfrac{1}{25}$, choice \textbf{A}.
\end{hint}
\end{problem}

\begin{problem}

\textbf{4.} Solve:
\[
  \left\{\begin{aligned}
    y &= -1-2x \\
    10x+5y &= -2
  \end{aligned}\right.
\]



A) $(2,-5)$

B) $(0,-1)$

C) No solution

D) $(1,-3)$

E) Dependent, $y=-1-2x$

\begin{hint}
\textbf{Solution.} Substitute $y=-1-2x$ into the second equation:
\[
  10x+5(-1-2x) = 10x-5-10x = -5,
\]
so the system requires $-5=-2$, which is false. The lines are parallel and there is
\textbf{no solution} (choice C).
\end{hint}
\end{problem}

\begin{problem}

\textbf{5.} Multiply: $(p-5)^{2}$.



A) $p^{2}-10p+25$

B) $p^{2}-10p-25$

C) $p^{2}-25$

D) $p^{2}+25$

E) $p^{2}+10p+25$

\begin{hint}
\textbf{Solution.} $(p-5)^{2}=p^{2}-2(5)p+25=p^{2}-10p+25$, choice \textbf{A}.
\end{hint}
\end{problem}

\section*{Free response (6--25)}

\begin{problem}

\textbf{6.} Solve by graphing.
\[
  \left\{\begin{aligned}
    -2x+3y &= -9 \\
    x-y &= 4
  \end{aligned}\right.
\]

\begin{hint}
\textbf{Solution.} Put each equation in slope-intercept form:
\[
  -2x+3y=-9 \;\Longrightarrow\; 3y=2x-9 \;\Longrightarrow\; y=\tfrac{2}{3}x-3,
\]
\[
  x-y=4 \;\Longrightarrow\; -y=-x+4 \;\Longrightarrow\; y=x-4 .
\]
\figbox{\gridsm}
\end{hint}
\end{problem}

\begin{problem}

\textbf{7.} Solve by substitution:
\[
  \left\{\begin{aligned}
    y &= 4x+15 \\
    y &= 2x+7
  \end{aligned}\right.
\]

\begin{hint}
\textbf{Solution.} Set the right-hand sides equal:
\[
  4x+15 = 2x+7 \;\Longrightarrow\; 2x = -8 \;\Longrightarrow\; x = -4 .
\]
Then $y = 2(-4)+7 = -1$. The solution is $(-4,-1)$.
\end{hint}
\end{problem}

\begin{problem}

\textbf{8.} Multiply: $(4uv)\left(7u^{2}v^{3}\right)\left(4u^{2}v\right)$.

\begin{hint}
\textbf{Solution.} Multiply the coefficients and add the exponents on each variable:
\[
  4\cdot 7\cdot 4 = 112, \qquad u^{1+2+2}=u^{5}, \qquad v^{1+3+1}=v^{5},
\]
so the product is $112u^{5}v^{5}$.
\end{hint}
\end{problem}

\begin{problem}

\textbf{9.} Subtract: $\left(9s^{3}-3s^{2}-5\right)-\left(-4s^{2}+3s-2\right)$.

\begin{hint}
\textbf{Solution.} Distribute the minus sign and combine like terms:
\[
  9s^{3}-3s^{2}-5+4s^{2}-3s+2 = 9s^{3}+s^{2}-3s-3 .
\]
\end{hint}
\end{problem}

\begin{problem}

\textbf{10.} Simplify: $\dfrac{-4t^{-3}u^{-9}}{-28tu^{3}}$.

\begin{hint}
\textbf{Solution.} Divide the coefficients and subtract the exponents:
\[
  \frac{-4}{-28}=\frac{1}{7}, \qquad t^{-3-1}=t^{-4}, \qquad u^{-9-3}=u^{-12},
\]
so the result is $\dfrac{1}{7t^{4}u^{12}}$.
\end{hint}
\end{problem}

\begin{problem}

\textbf{11.} Multiply: $4k^{3}\left(5k^{2}-5k+1\right)$.

\begin{hint}
\textbf{Solution.} $20k^{5}-20k^{4}+4k^{3}$.
\end{hint}
\end{problem}

\begin{problem}

\textbf{12.} Write the number $5.8\times 10^{-4}$ in decimal notation.

\begin{hint}
\textbf{Solution.} Move the decimal point four places left: $0.00058$.
\end{hint}
\end{problem}

\begin{problem}

\textbf{13.} An unladen European swallow flying against the wind traveled 36 mi in
2.5 h. Flying in the opposite direction, with the wind, the swallow was able to travel
the same distance in 1.5 h. Find the speed of the swallow in calm weather, and the rate
of the wind.

\begin{hint}
\textbf{Solution.} Let $r$ be the speed in calm air and $w$ the rate of the wind:
\[
  (r-w)(2.5)=36 \;\Longrightarrow\; r-w=14.4, \qquad
  (r+w)(1.5)=36 \;\Longrightarrow\; r+w=24 .
\]
Adding gives $2r=38.4$, so $r=19.2$ mph, and then $w=24-19.2=4.8$ mph.
\end{hint}
\end{problem}

\begin{problem}

\textbf{14.} Multiply: $(5y+9)(5y-9)$.

\begin{hint}
\textbf{Solution.} This is a difference of squares: $(5y)^{2}-9^{2}=25y^{2}-81$.
\end{hint}
\end{problem}

\begin{problem}

\textbf{15.} Solve by the addition method:
\[
  \left\{\begin{aligned}
    2x-5y &= -26 \\
    3x+4y &= 7
  \end{aligned}\right.
\]

\begin{hint}
\textbf{Solution.} Multiply the first equation by 4 and the second by 5:
\[
  8x-20y=-104, \qquad 15x+20y=35 .
\]
Adding gives $23x=-69$, so $x=-3$. Substituting into $3x+4y=7$:
\[
  3(-3)+4y=7 \;\Longrightarrow\; -9+4y=7 \;\Longrightarrow\; 4y=16 \;\Longrightarrow\; y=4 .
\]
The solution is $(-3,4)$.
\end{hint}
\end{problem}

\begin{problem}

\textbf{16.} Graph by using the slope and $y$-intercept: $y=\dfrac{2}{3}x-4$.

\begin{hint}
\textbf{Solution.} $m=\dfrac{2}{3}$ and the $y$-intercept is $-4$.
\figbox{\gridsm}
\end{hint}
\end{problem}

\begin{problem}

\textbf{17.} Compute the slope of the line containing the points $P_{1}(-2,1)$ and
$P_{2}(3,-1)$.

\begin{hint}
\textbf{Solution.}
\[
  m=\frac{y_{2}-y_{1}}{x_{2}-x_{1}}=\frac{-1-1}{3-(-2)}=\frac{-2}{5}.
\]
\end{hint}
\end{problem}

\textbf{Solution.} $7^{2}c^{-6}w^{-8}=\dfrac{49}{c^{6}w^{8}}$.
\end{hint}
\end{problem}

\begin{problem}

\textbf{19.} Multiply and write the answer in scientific notation:

\begin{problem}

\textbf{20.} Graph the solution set: $2x+3y \le -6$.

\begin{hint}
\textbf{Solution.} Graph the line $2x+3y=-6$, then test $(0,0)$:
\[
  2(0)+3(0) \le -6 \;\Longrightarrow\; 0 \le -6,
\]
which is false, so shade the side that does \emph{not} contain the origin.
\figbox{\gridsm}
\end{hint}
\end{problem}

\begin{problem}

\textbf{21.} Compute the equation of the line that passes through the points $(-3,-5)$
and $(3,13)$.

\begin{hint}
\textbf{Solution.}
\[
  m=\frac{13-(-5)}{3-(-3)}=\frac{18}{6}=3 .
\]
Using $y=mx+b$ with the point $(3,13)$: $13=3(3)+b$, so $b=4$ and $y=3x+4$.
\end{hint}
\end{problem}

\begin{problem}

\textbf{22.} Solve the system.
\[
  \left\{\begin{aligned}
    10x+16y &= 9 \\
    5x+48y &= 17
  \end{aligned}\right.
\]

\begin{hint}
\textbf{Solution.} Multiplying the second equation by $-2$ gives
$-10x-96y=-34$; adding it to the first eliminates $x$ and leaves $-80y=-25$, so
$y=\dfrac{25}{80}=\dfrac{5}{16}$. Substituting back,
\[
  10x+16\left(\tfrac{5}{16}\right)=9 \;\Longrightarrow\; 10x+5=9
  \;\Longrightarrow\; 10x=4 \;\Longrightarrow\; x=\tfrac{2}{5},
\]
giving the solution $\left(\tfrac{2}{5},\tfrac{5}{16}\right)$.
\end{hint}
\end{problem}

\begin{problem}
\textbf{23.} Multiply: $\left(6y^{3}+4y^{2}+6\right)(6y-1)$.

\textbf{Answer:} $\answer{36y^4+18y^3-4y^2+36y-6}$

\begin{hint}
\textbf{Solution.} Distribute each term:
\[
6y(6y^{3}+4y^{2}+6)=36y^{4}+24y^{3}+36y,\qquad -1(6y^{3}+4y^{2}+6)=-6y^{3}-4y^{2}-6 .
\]
Adding the like terms gives $36y^{4}+18y^{3}-4y^{2}+36y-6$.
\end{hint}
\end{problem}

\begin{problem}

\textbf{24.} A college football team, playing in the College Football Playoff, scored 34
points in seven-point touchdowns and three-point field goals. If all the field goals had
been touchdowns and all the touchdowns had been field goals, the team would have scored
26 points. Find how many touchdowns and how many field goals the team actually scored.

\begin{hint}
\textbf{Solution.} Let $t$ be touchdowns and $f$ field goals:
\[
  7t+3f=34, \qquad 3t+7f=26 .
\]
Adding gives $10t+10f=60$, so $t+f=6$; subtracting gives $4t-4f=8$, so $t-f=2$.
Hence $t=4$ touchdowns and $f=2$ field goals, which checks:
$7(4)+3(2)=34$ and $3(4)+7(2)=26$.
\end{hint}
\end{problem}

\begin{problem}

\textbf{25.} Divide: $\left(2x^{2}-2x+3\right)\div(x-2)$.

\begin{hint}
\textbf{Solution.} Long division gives quotient $2x+2$ with remainder $7$, that is
$2x+2+\dfrac{7}{x-2}$, since $(x-2)(2x+2)+7=2x^{2}-2x+3$.
\end{hint}
\end{problem}
