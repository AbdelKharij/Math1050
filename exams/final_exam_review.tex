\textbf{Directions.} You have 105 minutes to complete this exam. Each problem is worth

\begin{problem}

\textbf{1.} 3 is what percent of 20?

\begin{hint}
\textbf{Solution.} $\dfrac{3}{20}=0.15=15\%$.
\end{hint}
\end{problem}

\begin{problem}

\textbf{2.} Write the equation in the form $y=mx+b$: $-8x+5y=5$.

\begin{hint}
\textbf{Solution.} $5y=8x+5$, so $y=\dfrac{8}{5}x+1$.
\end{hint}
\end{problem}

\begin{problem}

\textbf{3.} A smart TV that regularly sells for \$1075 is on sale on Black Friday, $40\%$
off the regular price. Find the sale price.

\begin{hint}
\textbf{Solution.} The discount is $1075(0.40)=\$430$, so the sale price is
$1075-430=\$645$.
\end{hint}
\end{problem}

\begin{problem}

\textbf{4.} Simplify: $\left(-y^{4}\right)^{8}$.

\begin{hint}
\textbf{Solution.} The exponent 8 is even, so the result is positive:
$\left(-y^{4}\right)^{8}=y^{32}$.
\end{hint}
\end{problem}

\begin{problem}

\textbf{6.} Divide: $\dfrac{7x^{10}-3x^{8}-5x^{6}}{x^{4}}$.

\begin{hint}
\textbf{Solution.} Divide each term by $x^{4}$:
\[
  \frac{7x^{10}}{x^{4}}-\frac{3x^{8}}{x^{4}}-\frac{5x^{6}}{x^{4}}
  = 7x^{6}-3x^{4}-5x^{2}.
\]
\end{hint}
\end{problem}

\begin{problem}

\textbf{10.} Solve: $x^{2}+3x=0$.

\begin{hint}
\textbf{Solution.} Factor out $x$:
\[
  x(x+3)=0 \;\Longrightarrow\; x=0 \ \text{ or } \ x=-3 .
\]
\end{hint}
\end{problem}

\begin{problem}

\textbf{11.} Factor: $27b^{2}+6bx-45b-10x$.

\begin{hint}
\textbf{Solution.} Factor by grouping:
\[
  \left(27b^{2}-45b\right)+\left(6bx-10x\right)
  = 9b(3b-5)+2x(3b-5) = (3b-5)(9b+2x).
\]
\end{hint}
\end{problem}

\textbf{12.} Factor: $2x^{2}-9x-18$.

\begin{hint}
\textbf{Solution.} $2x^{2}-9x-18=(2x+3)(x-6)$.

\begin{problem}

\textbf{13.} Factor: $9x^{4}-99x^{3}+162x^{2}$.

\begin{hint}
\textbf{Solution.} Factor out the common $9x^{2}$, then factor the trinomial:
\[
  9x^{2}\left(x^{2}-11x+18\right) = 9x^{2}(x-2)(x-9).
\]
\end{hint}
\end{problem}

\begin{problem}

\textbf{15.} Divide: $\left(2x^{2}-3x-18\right)\div(x-4)$.

\begin{hint}
\textbf{Solution.} Long division gives quotient $2x+5$ with remainder 2, that is
$2x+5+\dfrac{2}{x-4}$, since $(x-4)(2x+5)+2=2x^{2}-3x-18$.

\begin{problem}

\textbf{16.} Solve: $2x+1>3x-5$.

\begin{hint}
\textbf{Solution.} Subtract $3x$ and 1 from both sides:
\[
  -x > -6 \;\Longrightarrow\; x < 6,
\]
the inequality reversing because both sides were divided by $-1$.
\end{hint}
\end{problem}

\begin{problem}

\textbf{18.} Solve: $\dfrac{4}{5}x-1=\dfrac{1}{5}x+14$.

\begin{hint}
\textbf{Solution.} Multiply through by 5, then collect terms:
\[
  4x-5=x+70 \;\Longrightarrow\; 3x=75 \;\Longrightarrow\; x=25 .
\]
\end{hint}
\end{problem}

\begin{problem}

\textbf{19.} Simplify: $10 \div \dfrac{2^{2}}{8-6}-(-7)$.

\begin{hint}
\textbf{Solution.}
\[
  10 \div \frac{4}{2}+7 = 10 \div 2+7 = 5+7 = 12 .
\]
\end{hint}
\end{problem}

\begin{problem}

\textbf{20.} Solve by the addition method:
\[
  \left\{\begin{aligned}
    2x-2y &= 10 \\
    4x-3y &= 17
  \end{aligned}\right.
\]

\begin{hint}
\textbf{Solution.} Multiply the first equation by $-2$ and add:
\[
  -4x+4y=-20, \qquad 4x-3y=17 \;\Longrightarrow\; y=-3 .
\]
Then $2x-2(-3)=10$, so $2x=4$ and $x=2$. The solution is $(2,-3)$.
\end{hint}
\end{problem}

\begin{problem}

\textbf{21.} Solve by substitution:
\[
  \left\{\begin{aligned}
    y &= 1-3x \\
    7x+4y &= 24
  \end{aligned}\right.
\]

\begin{hint}
\textbf{Solution.} Substitute $y=1-3x$:
\[
  7x+4(1-3x)=24 \;\Longrightarrow\; -5x+4=24 \;\Longrightarrow\; x=-4 .
\]
Then $y=1-3(-4)=13$. The solution is $(-4,13)$.
\end{hint}
\end{problem}

\begin{problem}

\textbf{22.} The Cool-r-We HVAC Shop charges \$2000 for an AC unit and \$35 per hour for
labor. If the total charge was \$3750, how many hours did the shop's technicians work?

\begin{hint}
\textbf{Solution.}
\[
  2000+35h=3750 \;\Longrightarrow\; 35h=1750 \;\Longrightarrow\; h=50 \text{ hours}.
\]
\end{hint}
\end{problem}

\begin{problem}

\textbf{23.} Find the equation of the line that passes through the points $(-2,6)$ and
$(2,-14)$. Write the equation in the form $y=mx+b$.

\begin{hint}
\textbf{Solution.}
\[
  m=\frac{-14-6}{2-(-2)}=\frac{-20}{4}=-5 .
\]
Using $(-2,6)$: $6=(-5)(-2)+b=10+b$, so $b=-4$ and $y=-5x-4$.
\end{hint}
\end{problem}

\begin{problem}

\textbf{24.} A restaurant's house Italian dressing is made by blending olive oil that
costs \$1.35 per cup with vinegar that costs \$0.35 per cup. How many cups of vinegar are
used if the cost of 10 cups of the blended dressing is \$0.65 per cup? Round your answer
to the nearest whole number.

\begin{hint}
\textbf{Solution.} Let $x$ be the cups of vinegar, so $10-x$ cups are olive oil:
\[
  1.35(10-x)+0.35x = 0.65(10)=6.50 .
\]
Then $13.5-x=6.5$, so $x=7$ cups of vinegar.

\end{image}
\end{hint}
\end{problem}
