\textbf{Directions.} There are 25 questions. Each question is worth 4 points. Please show
your work in order to obtain any partial credit.

\begin{problem}
\textbf{1.} Find the complement of a $30^{\circ}$ angle.

\textbf{Solution.} $90^{\circ}-30^{\circ}=60^{\circ}$.
\end{problem}

\begin{problem}
\textbf{2.} 15 is what percent of 60?

\textbf{Solution.} $\dfrac{15}{60}=0.25=25\%$.

\emph{Note: the scanned worksheet shows $60\times 0.15=9$, which answers the different
question ``what is 15\% of 60?''\ Check with your instructor which was intended.}
\end{problem}

\begin{problem}
\textbf{3.} Evaluate $\dfrac{-4bc}{2a-b}$ when $a=-2$, $b=4$, and $c=-1$.

\textbf{Solution.}
\[
  \frac{-4(4)(-1)}{2(-2)-4}=\frac{16}{-4-4}=\frac{16}{-8}=-2 .
\]

\emph{Note: the scanned worksheet records $-8$ for this one; $16\div(-8)=-2$.}
\end{problem}

\begin{problem}
\textbf{4.} Evaluate: $4(-8) \div\left[2(7-3)^{2}\right]$.

\textbf{Solution.} $4(-8)=-32$ and $2(7-3)^{2}=2(16)=32$, so the quotient is $-1$.
\end{problem}

\begin{problem}
\textbf{6.} Subtract: $3-(-12)-5$.

\textbf{Solution.} $3+12-5=15-5=10$.
\end{problem}

\begin{problem}
\textbf{7.} Multiply: $\left(\frac{5}{12}\right)\left(-\frac{8}{15}\right)\left(-\frac{1}{3}\right)$.

\textbf{Solution.} $\dfrac{5\cdot 8\cdot 1}{12\cdot 15\cdot 3}=\dfrac{40}{540}=\dfrac{2}{27}$,
positive because there are two negative factors.
\end{problem}

\begin{problem}
\textbf{8.} The temperature at which mercury boils is $360^{\circ}\mathrm{C}$. Mercury
freezes at $-39^{\circ}\mathrm{C}$. Find the difference between the temperature at which
mercury boils and the temperature at which it freezes.

\textbf{Solution.} $360-(-39)=360+39=399^{\circ}\mathrm{C}$.
\end{problem}

\begin{problem}
\textbf{9.} Find the least common multiple of 12, 42, and 14.

\textbf{Solution.} $12=2^{2}\cdot 3$, $42=2\cdot 3\cdot 7$, $14=2\cdot 7$, so the least
common multiple is $2^{2}\cdot 3\cdot 7=84$.
\end{problem}

\begin{problem}
\textbf{10.} Find the value of the third angle of a triangle whose other angles are
$20^{\circ}$ and $130^{\circ}$.

\textbf{Solution.}
\[
  x+130+20=180 \;\Longrightarrow\; x+150=180 \;\Longrightarrow\; x=30^{\circ}.
\]
\end{problem}

\begin{problem}
\textbf{11.} Translate into a variable expression: ``the product of a number and ten more
than the number.''

\textbf{Solution.} With $n$ the number, the expression is $n(n+10)$.

\end{problem}

\begin{problem}
\textbf{13.} Solve: $5x+9+2x=23$.

\textbf{Solution.}
\[
  7x+9=23 \;\Longrightarrow\; 7x=14 \;\Longrightarrow\; x=2 .
\]
\end{problem}

\begin{problem}
\textbf{14.} Solve: $\dfrac{x}{4}-6=1$.

\textbf{Solution.} $\dfrac{x}{4}=7$, so $x=28$.
\end{problem}

\begin{problem}
\textbf{15.} Solve: $2x-5=4(3x+1)-2$.

\textbf{Solution.}
\[
  2x-5=12x+4-2=12x+2 \;\Longrightarrow\; -7=10x \;\Longrightarrow\; x=-\frac{7}{10}=-0.7 .
\]
\end{problem}

\begin{problem}
\textbf{16.} Evaluate: $(-2)^{4}$.

\textbf{Solution.} $(-2)^{4}=16$, positive because the exponent is even.
\end{problem}

\begin{problem}
\textbf{17.} Evaluate: $-|4-5|$.

\textbf{Solution.} $|4-5|=|-1|=1$, so $-|4-5|=-1$.
\end{problem}

\begin{problem}
\textbf{18.} $25\%$ of what is 6?

\textbf{Solution.} $0.25x=6$, so $x=\dfrac{6}{0.25}=24$.
\end{problem}

\begin{problem}
\textbf{19.} Write $65\%$ as a fraction and a decimal.

\textbf{Solution.} Fraction: $\dfrac{65}{100}=\dfrac{13}{20}$. Decimal: $0.65$.
\end{problem}

\begin{problem}
\textbf{21.} At 5:00 am a train leaves a station travelling at 50 mph. At 6:00 am another
train leaves the same station on the same track, travelling in the same direction at
70 mph. At 7:00 am, how many miles separate the two trains?

\textbf{Solution.} By 7:00 am the first train has travelled $2(50)=100$ mi and the second
$1(70)=70$ mi, so they are $100-70=30$ mi apart.
\end{problem}

\begin{problem}
\textbf{22.} Find the perimeter of a triangle with sides 4 ft 5 in., 5 ft 3 in., and
6 ft 11 in.

\textbf{Solution.} The feet total $4+5+6=15$ ft and the inches total
$5+3+11=19$ in. $=1$ ft 7 in., so the perimeter is 16 ft 7 in.
\end{problem}

\begin{problem}
\textbf{23.} Solve for $x$: $\dfrac{2}{3}x-1=\dfrac{1}{3}x+5$.

\textbf{Solution.}
\[
  \frac{2}{3}x-\frac{1}{3}x-1=5 \;\Longrightarrow\; \frac{1}{3}x=6 \;\Longrightarrow\; x=18 .
\]
\end{problem}

\begin{problem}
\textbf{24.} Find the circumference of a circle whose radius is 5 in. Use 3.14 for $\pi$
and write all the digits of your answer.

\textbf{Solution.} $C=2\pi r=2(3.14)(5)=31.4$ in.

\emph{Note: the scanned worksheet shows $2\times 3.14\times 5^{2}=157$, which squares the
radius. That is the area formula $A=\pi r^{2}$ (and doubled); circumference uses
$2\pi r$.}
\end{problem}

\begin{problem}
\textbf{25.} Simplify: $5(x+6)-3(x-2)$.

\textbf{Solution.}
\[
  5x+30-3x+6 = 2x+36 .
\]
\end{problem}
